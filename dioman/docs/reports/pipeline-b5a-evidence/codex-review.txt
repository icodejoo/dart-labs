/// 位掩码枚举；合成依赖表也走同一实现。
List<int> subsets(int size, [Map<int, Set<int>> needs = const {}]) => [
      for (var mask = 0; mask < 1 << size; mask++)
        if (needs.entries.every((e) =>
            mask & (1 << e.key) == 0 ||
            e.value.every((i) => mask & (1 << i) != 0)))
          mask
    ];

      final probe = request.path.startsWith(mockUrl);
      final hit = ++realHits;
      final status = probe

          : flow == PowerFlow.http || flow == PowerFlow.flaky && hit <= 2
              ? 500
              : 200;
      if (probe) realHits--;
      final gate = Completer<void>();
      clock.createTimer(tick, gate.complete);
      net.routes['${request.method}:${request.path}'] = [

          body: {
            flowCode: flow == PowerFlow.auth &&

                : 0,
            flowData: null
          },
          error: !probe && flow == PowerFlow.network
              ? DioException(
                  requestOptions: request,
                  type: DioExceptionType.connectionError)
              : null,
          gate: gate.future,
        )
      ];
    });
    h = InvariantHarness(net, clock);
    api = testClient(
        network: net, clock: clock, plugins: [...h.probes(), ...plugins()]);
    api.dio.transformer = B3Transformer();
  }

  final int mask;
  final PowerFlow flow;
  final clock = FakeClock();
  final tokens = TestTokens();
  final connectivity = StreamController<bool>.broadcast(sync: true);
  late final FakeNetwork net;
  late final InvariantHarness h;
  late final Dioman<Result> api;
  final requests = <Future<Result>>[];

  /// 调用方刚离开时的订阅快照，避免最终 dispose 清表掩盖首发泄漏。
  final leavingSubscriptions = <int>[];
  bool online = true;
  int realHits = 0;

  /// 查询本次安装表。
  bool has(String name) => mask & (1 << interactionNames.indexOf(name)) != 0;

  /// 创建本次独占的插件对象。
  List<DiomanPlugin<Result>> plugins() {
    final factories = <String, DiomanPlugin<Result> Function()>{
      loadingName: () => h.loadingPlugin(delayClose: tick),
      notifyName: () => h.notifyPlugin(),
      retryName: () =>
          DiomanRetry<Result>(max: 2, delay: (_, __, ___) => Duration.zero),
      breakerName: () => DiomanBreaker<Result>(failureThreshold: 2),
      authName: () => b3Auth(tokens,
          refresh: h.refreshInRound(0, (tm, r) async {
            final gate = Completer<void>();
            clock.createTimer(tick, gate.complete);
            await gate.future;
            tm.setTokens(accessToken: b3Fresh);
          })),
      mockName: () => DiomanMock<Result>(mockUrl: mockUrl),
      shareName: () => DiomanShare<Result>(),
      cacheName: () => DiomanCache<Result>(),
      offlineName: () => DiomanOffline<Result>(
          isOnline: () => online, onConnectivityChanged: connectivity.stream),
    };
    return [
      for (final name in interactionNames)
        if (has(name)) factories[name]!()
    ];
  }

  /// 登记嵌套载荷与逻辑请求标记；同 key 请求不靠标记参与构键。
  void launch({String key = flowKey, CancelToken? token}) {
    final id = requests.length + 1;
    requests.add(h.track(
        id,
        api
            .send<dynamic>(flowGet, flowPath,
                data: {
                  flowItems: [
                    {flowN: 0}
                  ]
                },
                call: DiomanCall(key: key, cancelToken: token, headers: {
                  invariantRequestHeader: id
                }, options: {
                  DiomanShare.option: true,
                  DiomanCache.option: true,
                }))
            .whenComplete(() {
          leavingSubscriptions.add(h.contexts
              .singleWhere((ctx) => ctx.id == id)
              .cancellationSubscriptions);
        })));
  }

  /// 让网络到达阻塞点，只推进微任务和事件队列。
  Future<void> turn() async {
    await b3Turn();
    await b3Turn();
  }

  /// 有上限地驱动假时间；超出预算明确失败而不是永久挂起。
  Future<void> drain() async {
    var done = false;
    final pending = Future.wait(requests);
    pending.then<void>((_) => done = true,
        onError: (Object e, StackTrace s) => done = true);
    for (var i = 0; i < 100 && !done; i++) {
      await turn();
      clock.elapse(tick);
    }
    expect(done, isTrue, reason: '假时钟流程必须有界落定');
    await pending;
    for (var i = 0; i < 4; i++) {
      await turn();
      clock.elapse(tick);
    }
  }

  /// 九条流程只发请求、改变连接状态、取消和推进假时钟。
  Future<void> run() async {
    switch (flow) {
      case PowerFlow.success:
        launch();
        await drain();
        launch();
      case PowerFlow.auth:
        if (has(authName)) h.beginRefreshRound(0);
        launch();
        launch();
        launch(key: 'other');
      case PowerFlow.concurrent:
        launch();
        launch();
      case PowerFlow.cancelAll:
        launch();
        launch();
        await turn();
        api.cancelAll();
      case PowerFlow.cancelLeader:
        final token = CancelToken();
        launch(token: token);
        launch();
        await turn();
        token.cancel();
      case PowerFlow.recovery:
        online = false;
        launch();
        launch();
        await turn();
        clock.elapse(Duration.zero);
        online = true;
        connectivity.add(true);
      case PowerFlow.http:
      case PowerFlow.flaky:
      case PowerFlow.network:
        launch();
    }
    await drain();
  }

  /// 正常收尾即检查内部诊断，再清缓存与销毁，最后再检查一次。
  Future<void> check() async {
    h.assertAll(api);
    expect(leavingSubscriptions, everyElement(0), reason: 'I3 离开当时已摘监听');
    final cache = api.plugin<DiomanCacheApi>();
    final share = api.plugin<DiomanShareApi>();
    expect(share?.activeEntries ?? 0, 0);
    expect(cache?.pendingWrites ?? 0, 0);
    await cache?.clear();
    expect(cache?.entryCount ?? 0, 0);
    expect(cache?.pendingWrites ?? 0, 0);
    api.dispose();
    expect(share?.activeEntries ?? 0, 0);
    expect(cache?.entryCount ?? 0, 0);
    expect(cache?.pendingWrites ?? 0, 0);
  }

  /// 失败时也释放流和客户端。
  Future<void> dispose() async {
    api.dispose();
    await connectivity.close();
  }
}

def inspect(path):
    """拒绝缺文件、重复 SF 和未覆盖计数；纯声明单独列明，不宣称空分母百分比。"""
    files = {p.relative_to(SOURCE).as_posix(): p for p in SOURCE.rglob('*.dart')}
    declarations = {
        'token_manager.dart': 'abstract interface class DiomanTokenManager',
        'cache_store.dart': 'abstract interface class DiomanCachePersist',
        'stages.dart': 'const diomanBuiltinStages',
    }
    found = {}
    for block in path.read_text(encoding='utf-8').split('end_of_record'):
        lines = block.splitlines()
        source = next((s[3:].replace('\\', '/') for s in lines if s.startswith('SF:')), '')
        marker = '/lib/src/pipeline/'
        if marker not in source:
            continue
        name = source.split(marker)[1]
        assert name in files and name not in found, name
        rows = [s[3:].split(',') for s in lines if s.startswith('DA:')]
        branches = [s[5:].split(',') for s in lines if s.startswith('BRDA:')]
        counts = [len(rows), sum(int(r[1]) > 0 for r in rows),
                  len(branches), sum(r[3] != '-' and int(r[3]) > 0 for r in branches)]
        found[name] = counts
        missed = [r for r in rows if int(r[1]) == 0]
        missed += [r for r in branches if r[3] == '-' or int(r[3]) == 0]
        assert not missed, (name, missed)
        for key, count in zip(('LF', 'LH', 'BRF', 'BRH'), counts):
            summary = [int(s.split(':')[1]) for s in lines if s.startswith(key + ':')]
            assert not summary or summary == [count], (name, key)
    assert set(files) - set(found) <= set(declarations), set(files) - set(found)
    for name, marker in declarations.items():
        source = files[name].read_text(encoding='utf-8')
        assert marker in source
        # 与磁盘全部源文件一一对照；声明文件不得偷偷产生未计数的方法体。
        code = '\n'.join(s for s in source.splitlines() if not s.lstrip().startswith('///'))
        assert '=>' not in code and 'return ' not in code, name
        found.setdefault(name, [0, 0, 0, 0])
    assert set(found) == set(files)
    print('file LF LH BRF BRH')
    for name, counts in sorted(found.items()):
        assert name in declarations or counts[0] > 0, name
        print(name, *counts)
    (DEST / 'coverage.json').write_text(json.dumps(found, indent=2) + '\n', encoding='utf-8')
    print(f'{len(files)} source files accounted for; all executable lines/branches 100%; PASS')
    print('cache_store/token_manager: abstract contracts; stages: enums/const tables; no executable denominator')

    for name, module, old, new in CASES:
        path = ROOT / f'lib/src/pipeline/{module}.dart'
        original = path.read_bytes()
        digest = hashlib.sha256(original).hexdigest()
        source = original.decode('utf-8').replace('\r\n', '\n')
        assert source.count(old) == 1, (name, source.count(old))
        start = time.perf_counter()
        try:
            path.write_text(source.replace(old, new), encoding='utf-8')
            proc = subprocess.run([shutil.which('dart'), 'test', TEST,
                                   '--reporter', 'json', '--timeout', '10s'],
                                  cwd=ROOT, capture_output=True, text=True,
                                  encoding='utf-8', errors='replace', timeout=120)
            output = proc.stdout + proc.stderr
            (DEST / f'mutation-{name}.jsonl').write_text(output, encoding='utf-8')
            events = [json.loads(s) for s in proc.stdout.splitlines() if s.startswith('{')]
            failures = [e for e in events if e.get('type') == 'testDone'
                        and e.get('result') in ('failure', 'error')]
            tests = {e['test']['id']: e['test']['name'] for e in events if e.get('type') == 'testStart'}
            powerset = [e for e in failures if ' mask=' in tests[e['testID']]]
            assert proc.returncode != 0 and powerset, name
            assert not any(s in output for s in ('Failed to load', 'Compilation failed',
                                                  'Test timed out')), name
        finally:
            path.write_bytes(original)
        restored = hashlib.sha256(path.read_bytes()).hexdigest()
        assert restored == digest, name
        results.append(dict(mutation=name, source=str(path.relative_to(ROOT)),
                            test=TEST, failures=len(failures), powerset_failures=len(powerset),
                            baseline_failures=len(failures)-len(powerset),
                            seconds=round(time.perf_counter()-start, 3),
                            before_sha256=digest, restored_sha256=restored))
        (DEST / 'mutations.json').write_text(json.dumps(results, ensure_ascii=False,
                                                      indent=2)+'\n', encoding='utf-8')
        print(f'{name}: powerset failures={len(powerset)}; source restored; PASS', flush=True)

/// 启用 I1–I7，刷新按调用方声明的轮次统计。
class InvariantHarness {
  /// 绑定网络和时钟。
  InvariantHarness(this.network, this.clock);

  /// 网络发送证据。
  final FakeNetwork network;

  /// 时间资源证据。
  final FakeClock clock;

  /// 外层进入次数。
  final entries = <int, int>{};

  /// 外层离开次数。
  final exits = <int, int>{};

  /// 调用方落定次数。
  final settlements = <int, int>{};

  /// I2 回调配对序列。
  final loading = <Object, List<bool>>{};

  /// I4 每请求通知次数。
  final notifications = <int, int>{};

  /// I5 每轮刷新次数。
  final refreshRounds = <int, int>{};

  /// 声明即将开始的刷新轮，零调用也会被 I5 拒绝。
  void beginRefreshRound(int round) => refreshRounds[round] = 0;

  /// 包装真实 onRefresh；同一轮多次刷新会被 I5 拒绝。
  DiomanAuthRefresh refreshInRound(int round, DiomanAuthRefresh refresh) =>
      (tokens, result) {
        refreshRounds.update(round, (n) => n + 1);
        return refresh(tokens, result);
      };

  /// 各层保存的上下文，用于请求结束后追踪取消订阅。
  final contexts = <DiomanContext<Result>>[];

  /// 创建按回调身份记录 I2 的真实 loading 插件。
  DiomanLoading<Result> loadingPlugin(
      {Duration delay = Duration.zero, Duration delayClose = Duration.zero}) {
    final identity = Object();
    return DiomanLoading<Result>(
        onChanged: (visible) {
          loading.putIfAbsent(identity, () => []).add(visible);
        },
        delay: delay,
        delayClose: delayClose);
  }

  /// stringify 以 ctx.id 标记，notify 按逻辑请求而非响应身份记录 I4。
  DiomanNotify<Result> notifyPlugin() => DiomanNotify<Result>(
      stringify: (r, ctx) =>
          r.ok || r.kind == DiomanKind.cancel || ctx.request.silent
              ? ''
              : '${ctx.id}',
      notify: (message, r) {
        notifications.update(int.parse(message), (n) => n + 1,
            ifAbsent: () => 1);
      });

  /// 生成最外层及每阶段 pre/post 的直通探针。
  List<DiomanPlugin<Result>> probes() => [
        TestPlugin('invariant',
            stage: DiomanStage.outer,
            enforce: DiomanEnforce.pre, handler: (ctx, next) async {
          entries.update(ctx.id, (n) => n + 1, ifAbsent: () => 1);
          contexts.add(ctx);
          final r = await next();
          exits.update(ctx.id, (n) => n + 1, ifAbsent: () => 1);
          return r;
        }),
        for (final stage in DiomanStage.values)
          for (final enforce in DiomanEnforce.values)
            TestPlugin('probe:${stage.name}:${enforce.name}',
                stage: stage, enforce: enforce, handler: (ctx, next) async {
              final r = await next();
              expect(r.kind, isNot(DiomanKind.internal),
                  reason: 'I6 next 不抛且受控流程无 internal');
              return r;
            }),
      ];

  /// 登记一个逻辑调用及其最终一次落定。
  Future<T> track<T>(int id, Future<T> pending) => pending.whenComplete(() {
        settlements.update(id, (n) => n + 1, ifAbsent: () => 1);
      });

  /// 统一断言不变量；测试夹具自身的坏证据也必须被拒绝。
  void assertAll(Dioman<Result> api) {
    expect(entries, isNotEmpty, reason: 'I1 不允许空跑');
    expect(entries.keys.toSet(), exits.keys.toSet());
    expect(entries.keys.toSet(), settlements.keys.toSet());
    for (final id in entries.keys) {
      expect([entries[id], exits[id], settlements[id]], [1, 1, 1]);
    }
    expect(api.pending, 0, reason: 'I3');
    expect(api.cancellationSubscriptions, 0);
    expect(contexts.every((c) => c.cancellationSubscriptions == 0), isTrue);
    expect(clock.pendingTimers, 0);
    final bodies = <Object?, String>{};
    for (final send in network.sends) {
      if (send.invariant == null) continue;
      final body = jsonEncode(send.body);
      expect(body, bodies.putIfAbsent(send.invariant, () => body),
          reason: 'I7');
    }
    for (final sequence in loading.values) {
      expect(sequence.length.isEven, isTrue);
      for (var i = 0; i < sequence.length; i++) {
        expect(sequence[i], i.isEven);
      }
    }
    expect(notifications.values.every((n) => n <= 1), isTrue);
    expect(refreshRounds.values.every((n) => n == 1), isTrue);
  }
}

/// 捕获发送时快照，避免 Transformer 后续原地修改记录。
class FakeSend {
  /// 捕获实际有效配置与线上请求体。
  FakeSend(this.options, this.body, this.invariant);

  /// 实际 Dio 请求配置。
  final RequestOptions options;

  /// 发送时解码后的请求体。
  final Object? body;

  /// 逻辑请求标记。
  final Object? invariant;

/// 按方法路径路由、序列和第 n 次失败的假 adapter。
class FakeNetwork implements HttpClientAdapter {
  /// 创建空路由 adapter，缺省返回成功信封。
  FakeNetwork({this.onSend});

  /// 发送已记录时通知测试，避免猜测微任务次数。
  final void Function(FakeSend send)? onSend;

  /// 路由表，例如 GET:/pets。
  final routes = <String, List<FakeReply>>{};

  /// 每个路由的已发送次数。
  final counts = <String, int>{};

  /// 第 n 次发送覆盖脚本。
  final failOn = <int, Object>{};

  /// 所有实际发送。
  final sends = <FakeSend>[];

  /// 是否已关闭。
  bool closed = false;

  /// 执行假网络发送，读取 Dio 已转换的线上请求体。
  @override
  Future<ResponseBody> fetch(RequestOptions options,
      Stream<Uint8List>? requestStream, Future<void>? cancelFuture) async {
    final bytes = requestStream == null
        ? <int>[]
        : await requestStream.expand((b) => b).toList();
    Object? body;
    if (bytes.isNotEmpty) {
      final text = utf8.decode(bytes, allowMalformed: true);
      try {
        body = jsonDecode(text);
      } catch (_) {
        body = text;
      }
    }
    sends.add(FakeSend(options, body, options.headers[invariantRequestHeader]));
    final sendIndex = sends.length;
    onSend?.call(sends.last);
    final route = '${options.method}:${options.path}';
    final index = counts.update(route, (n) => n + 1, ifAbsent: () => 1) - 1;
    final replies = routes[route] ?? const [FakeReply()];
    final reply = replies[index < replies.length ? index : replies.length - 1];
    if (reply.gate != null) await reply.gate;
    final error = failOn[sendIndex] ?? reply.error;
    if (error != null) throw error;
    return ResponseBody.fromString(jsonEncode(reply.body), reply.status,
        headers: {
          Headers.contentTypeHeader: [Headers.jsonContentType]
        });
  }

  /// 记录 adapter 被关闭。
  @override
  void close({bool force = false}) => closed = true;
}

  final bool Function(DiomanResponse<dynamic>)? shouldStore;

  /// 存储层覆盖。
  final DiomanCachePolicy? cachePolicy;
}

/// 缓存管理能力；通过 `api.plugin<DiomanCacheApi>()` 获取。
abstract interface class DiomanCacheApi {
  /// 删除 [key] 并作废该键在途写入；等待持久化删除完成。
  Future<void> remove(String key);

  /// 清空缓存并立即作废所有在途写入；返回持久化清理完成信号。
  Future<void> clear();

  /// 在途屏障数，只读诊断；所有请求结束后为零。
  int get pendingWrites;

  /// 内存物理条目数，包含尚未惰性过期的条目。
  int get entryCount;
}

  final bool Function(DiomanRequest)? shouldCache;

  final bool Function(DiomanResponse<dynamic>)? shouldStore;

/// 内存缓存条目。
class _Entry {
  /// 保存响应及绝对到期时刻。
  _Entry(this.response, this.expiresAt);

  /// 来源响应。
  final DiomanResponse<dynamic> response;

  /// epoch 毫秒。
  final int expiresAt;
}

/// 在途期间的已写入顺序屏障。
class _Barrier {
  /// 当前在途数量。
  int pending = 0;

  /// 已发布的最大序号。
  int latest = 0;
}

/// 每客户端内存和串行持久化通道。
class _Cache<R extends DiomanResponse<dynamic>> extends DiomanPluginInstance<R>
    implements DiomanCacheApi {
  /// 保存存储与配置。
  _Cache(this.persist, this.maxEntries, this.clock, this.base);

  /// TTL 时钟。
  final DiomanClock clock;

  /// 内存容量。
  final int maxEntries;

  /// 外部持久化后端。
  final DiomanCachePersist? persist;

  /// 基础选项。
  final DiomanCacheOptions base;

  /// 内存条目。
  final entries = <String, _Entry>{};

  /// 只在在途期间存在的屏障。
  final barriers = <String, _Barrier>{};

  /// 单调请求序号。
  int sequence = 0;

  /// 最近 clear 的失效下界。
  int floor = 0;

  /// 所有落盘变更串行，clear 不会被旧 write 越过。
  Future<void> tail = Future<void>.value();

  /// 数据字段名，兼容旧存储。
  static const dataField = 'data';

  /// 到期字段名。
  static const expiresField = 'expiresAt';

  /// 新格式保存归一化元信息，旧格式仍按原始 HTTP 200 读取。
  static const resultField = 'result';

  /// 结果类别字段。
  static const kindField = 'kind';

  /// 逻辑状态字段。
  static const statusField = 'status';

  /// 原始 HTTP 状态字段。
  static const httpField = 'httpStatus';

  /// 业务码字段。
  static const codeField = 'code';

  /// 消息字段。
  static const messageField = 'message';

  /// 细分原因字段。
  static const reasonField = 'reason';

  /// 从新格式恢复归一化结果，老格式继续交给工厂识别信封。
  DiomanResponse<dynamic> restore(DiomanContext<R> ctx, Map saved) {
    final meta = saved[resultField];
    if (meta is! Map) {
      return ctx.factory.normalize(ctx.request,
          raw: Response<dynamic>(
              requestOptions: ctx.request.toRequestOptions(),
              statusCode: 200,
              data: saved[dataField]));
    }
    final kind = DiomanKind.values.byName(meta[kindField] as String);
    final init = DiomanResponse<dynamic>(
        ok: kind == DiomanKind.ok,
        kind: kind,
        status: meta[statusField] as int,
        httpStatus: meta[httpField] as int,
        code: meta[codeField],
        message: meta[messageField] as String,
        reason: meta[reasonField] as String?,
        data: meta[dataField],
        request: ctx.request);
    return ctx.factory.derive(init, init.data);
  }

  /// 在途屏障诊断。
  @override
  int get pendingWrites => barriers.length;

  /// 内存条目诊断。
  @override
  int get entryCount => entries.length;

  /// 写入并按最近使用顺序淘汰，屏障不随淘汰删除。
  void put(String key, _Entry entry) {
    entries.remove(key);
    entries[key] = entry;
    if (maxEntries > 0) {
      while (entries.length > maxEntries) {
        entries.remove(entries.keys.first);
      }
    }
  }

  /// 入队持久化变更；异常交给调用方，同时保持通道可继续使用。
  Future<void> enqueue(Future<void> Function() action) {
    final pending = tail.then((_) => action());
    tail = pending.then((_) {}, onError: (Object _, StackTrace __) {});
    return pending;
  }

  /// 删除也抬高本键屏障，阻止旧响应复活。
  @override
  Future<void> remove(String key) {
    entries.remove(key);
    barriers[key]?.latest = ++sequence;
    return persist == null
        ? Future<void>.value()
        : enqueue(() => persist!.remove(key));
  }

  /// 同步失效，随后串行清理后端。
  @override
  Future<void> clear() {
    floor = sequence;
    entries.clear();
    return persist == null
        ? Future<void>.value()
        : enqueue(() => persist!.erase());
  }

  /// 递归复制 Map/List；其余载荷按引用保留。
  Object? deep(Object? data) {
    if (data is Map) return data.map((k, v) => MapEntry(k, deep(v)));
    if (data is List) return data.map(deep).toList();
    return data;
  }

  /// 数据副本经扩展工厂创建，保留所有响应元信息。
  DiomanResponse<dynamic> clone(
      DiomanContext<R> ctx, DiomanResponse<dynamic> r, Object strategy) {
    if (strategy == DiomanClonePolicy.none) return r;
    final data = r.data;
    Object? copied;
    if (strategy is Object? Function(Object?)) {
      copied = strategy(data);
    } else if (strategy == DiomanClonePolicy.deep) {
      copied = deep(data);
    } else if (strategy == DiomanClonePolicy.shallow) {
      copied = data is Map
          ? Map.of(data)
          : data is List
              ? List.of(data)
              : data;
    } else {
      throw ArgumentError.value(strategy, 'clone');
    }
    return ctx.factory.derive(r, copied);
  }

  /// 查缓存、登记屏障、只写入仍新鲜的结果，并在 finally 回收。
  @override
  Future<DiomanResponse<dynamic>> handle(
      DiomanContext<R> ctx, DiomanNext<R> next) async {
    final o = resolveOption<DiomanCacheOptions>(
        ctx.request.call.options[DiomanCache.option],
        base: base,
        optIn: true,
        shorthand: (v) => v is Duration ? DiomanCacheOptions(expires: v) : null,
        merge: (a, b) => DiomanCacheOptions(
            enabled: b.enabled,
            expires: b.expires ?? a.expires,
            clone: b.clone ?? a.clone,
            shouldCache: b.shouldCache ?? a.shouldCache,
            shouldStore: b.shouldStore ?? a.shouldStore,
            cachePolicy: b.cachePolicy ?? a.cachePolicy));
    if (o == null || o.enabled == false || ctx.get(sourceProbe) == true) {
      return next();
    }
    if (!(o.shouldCache?.call(ctx.request) ??
        sourceSafeMethods.contains(ctx.request.method))) {
      return next();
    }
    final key = ctx.key();
    if (key == null) return next();
    final memo = o.cachePolicy != DiomanCachePolicy.persist;
    final disk = o.cachePolicy != DiomanCachePolicy.memo;
    if (disk && persist == null) throw StateError('persist/both 策略需要 persist');
    final barrier = barriers.putIfAbsent(key, _Barrier.new);
    barrier.pending++;
    final ticket = ++sequence;
    bool valid() => ticket > floor && ticket > barrier.latest;
    try {
      var hit = memo ? entries[key] : null;
      if (hit != null && hit.expiresAt <= clock.now().millisecondsSinceEpoch) {
        entries.remove(key);
        hit = null;
      }
      if (hit == null && disk) {
        await ctx.wait(tail);
        final cancel = ctx.cancelled();
        if (cancel != null) return cancel;
        final saved = await ctx.wait(Future.sync(() => persist!.read(key)));
        final cancelled = ctx.cancelled();
        if (cancelled != null) return cancelled;
        if (saved is Map && saved[expiresField] is int) {
          final expiry = saved[expiresField] as int;
          if (expiry > clock.now().millisecondsSinceEpoch && valid()) {
            hit = _Entry(restore(ctx, saved), expiry);
            if (memo) put(key, hit);
          } else {
            await ctx.wait(enqueue(() async {
              if (valid()) await persist!.remove(key);
            }));
            final cancel = ctx.cancelled();
            if (cancel != null) return cancel;
          }
        }
      }
      if (hit != null) {
        if (memo) put(key, hit);
        return clone(ctx, hit.response, o.clone!);
      }
      final r = await next();
      final cancel = ctx.cancelled();
      if (cancel != null) return cancel;
      var stored = false;
      if (valid() && (o.shouldStore?.call(r) ?? r.ok)) {
        stored = true;
        barrier.latest = ticket;
        final entry = _Entry(clone(ctx, r, o.clone!),
            clock.now().add(o.expires!).millisecondsSinceEpoch);
        if (memo) put(key, entry);
        if (disk) {
          await ctx.wait(enqueue(() => persist!.write(key, {
                dataField: r.raw?.data ?? r.data,
                expiresField: entry.expiresAt,
                resultField: {
                  kindField: r.kind.name,
                  statusField: r.status,
                  httpField: r.httpStatus,
                  codeField: r.code,
                  messageField: r.message,
                  reasonField: r.reason,
                  dataField: entry.response.data
                }
              })));
          final cancel = ctx.cancelled();
          if (cancel != null) return cancel;
        }
      }
      return r.ok || stored ? clone(ctx, r, o.clone!) : r;
    } finally {
      if (--barrier.pending == 0) barriers.remove(key);
    }
  }

  /// 作废在途写入并清空自有内存，外部持久化后端不归插件销毁。
  @override
  void dispose() {
    floor = sequence;
    entries.clear();
  }
}

/// 同 key 并发结果归属策略；关闭使用请求级 false。
enum DiomanSharePolicy {
  /// 首发共享一次传输。
  start,

  /// 最新发起的结果胜出。
  end,

  /// 首个成功胜出，全部失败取最后一个。
  race,

  /// 首发在内部重试，共享最终结果。
  retry,
}

/// 共享活动表诊断；例如 `api.plugin<DiomanShareApi>()!.activeEntries`。
abstract interface class DiomanShareApi {
  /// 四种策略当前活动轮次总数，结束或销毁后为零。
  int get activeEntries;
}

/// 请求级共享配置；null 字段沿用插件配置。
class DiomanShareOptions {
  /// 例如 `DiomanShareOptions(policy: DiomanSharePolicy.race)`。
  const DiomanShareOptions(
      {this.enabled, this.policy, this.retries, this.interval, this.methods});

  /// 本次开关。
  final bool? enabled;

  /// 归属策略。
  final DiomanSharePolicy? policy;

  /// 内部重试次数。
  final int? retries;

  /// 内部重试间隔。
  final Duration? interval;

  /// 允许共享的方法。
  final Set<String>? methods;
}

/// 请求级开启的共享插件；跟随者得到同一个只读结果。

  final DiomanSharePolicy policy;

  final int retries;

  /// retry 策略的等待间隔。
  final Duration interval;

  /// 方法硬白名单。
  final Set<String> methods;

  @override
  final bool enabled;

/// 一轮共享，不含逻辑请求的结算登记。
class _Shared {
  /// 保留与客户端工厂一致的销毁结果。
  _Shared(this.cancelled);

  /// 销毁时发布给尚在等待的订阅者。
  final DiomanResponse<dynamic> cancelled;

  /// 共享结果。
  final result = Completer<DiomanResponse<dynamic>>();

  /// 首发或 retry 的独立传输令牌。
  final token = CancelToken();

  /// 共享传输使用独立令牌并关闭 Dio 阶段超时，由订阅者各自限时。
  RequestPatch get transportPatch => RequestPatch(
      cancelToken: token,
      connectTimeout: Duration.zero,
      receiveTimeout: Duration.zero,
      sendTimeout: Duration.zero);

  /// 订阅者数。
  int subscribers = 0;

  /// 发起序号。
  int sequence = 0;

  /// 未结束的传输数。
  int pending = 0;
}

/// 每客户端按策略隔离的活动表。
class _Share<R extends DiomanResponse<dynamic>> extends DiomanPluginInstance<R>
    implements DiomanShareApi {
  /// 保存配置。
  _Share(this.base);

  /// 预合并配置。
  final DiomanShareOptions base;

  /// 每种策略一张表。
  final tables = {
    for (final p in DiomanSharePolicy.values) p: <String, _Shared>{}
  };

  /// 活动轮次总数。
  @override
  int get activeEntries =>
      tables.values.fold(0, (n, table) => n + table.length);

  /// 仅移除仍属于本轮的条目。
  void detach(Map<String, _Shared> table, String key, _Shared entry) {
    if (identical(table[key], entry)) table.remove(key);
  }

  /// 结果发布前先摘表，外层 retry 可立即重新入表。
  void finish(Map<String, _Shared> table, String key, _Shared entry,
      DiomanResponse<dynamic> r) {
    if (entry.result.isCompleted) return;
    detach(table, key, entry);
    entry.result.complete(r);
  }

  /// 订阅者整体上限为连接与接收之和；有请求体时再加发送阶段。
  /// 任一参与阶段为 null 或非正数时不限时，不模拟 Dio 的阶段进度。
  Duration? timeout(DiomanRequest request) {
    var total = Duration.zero;
    for (final stage in [
      request.connectTimeout,
      request.receiveTimeout,
      if (request.data != null) request.sendTimeout
    ]) {
      if (stage == null || stage <= Duration.zero) return null;
      total += stage;
    }
    return total;
  }

  /// 首发传输脱离首发调用方，最后一个订阅者离开才取消。
  Future<void> run(
      DiomanContext<R> ctx,
      DiomanNext<R> next,
      DiomanShareOptions o,
      Map<String, _Shared> table,
      String key,
      _Shared entry) async {
    final shared = ctx.derive(entry.transportPatch);
    ctx.state.retain();
    try {
      var r = await next(entry.transportPatch);
      if (o.policy == DiomanSharePolicy.retry) {
        for (var i = 0;
            i < o.retries! &&
                !r.ok &&
                r.kind != DiomanKind.internal &&
                !shared.isCancelled;
            i++) {
          if (o.interval! > Duration.zero) {
            await shared.wait(Completer<void>().future, timeout: o.interval);
          }
          if (shared.isCancelled) break;
          r = await next(entry.transportPatch);
        }
      }
      finish(table, key, entry, r);
    } catch (error, stack) {
      finish(
          table,
          key,
          entry,
          ctx.fail(DiomanFailure(
              kind: DiomanKind.internal, error: error, errorStack: stack)));
    } finally {
      ctx.state.release();
    }
  }

  /// end/race 的底层请求各自独立，订阅者离开不会取消其它传输。
  Future<void> compete(
      DiomanContext<R> ctx,
      DiomanNext<R> next,
      DiomanShareOptions o,
      Map<String, _Shared> table,
      String key,
      _Shared entry) async {
    final sequence = ++entry.sequence;
    entry.pending++;
    ctx.state.retain();
    try {
      final r = await next();
      entry.pending--;
      if (o.policy == DiomanSharePolicy.end
          ? sequence == entry.sequence
          : r.ok || entry.pending == 0) {
        finish(table, key, entry, r);
      }
    } finally {
      ctx.state.release();
    }
  }

  /// 解析 optIn 后订阅；wait 自行清理取消监听和计时器。
  @override
  Future<DiomanResponse<dynamic>> handle(
      DiomanContext<R> ctx, DiomanNext<R> next) async {
    final o = resolveOption<DiomanShareOptions>(
        ctx.request.call.options[DiomanShare.option],
        base: base,
        optIn: true,
        shorthand: (v) =>
            v is DiomanSharePolicy ? DiomanShareOptions(policy: v) : null,
        merge: (a, b) => DiomanShareOptions(
            enabled: b.enabled,
            policy: b.policy ?? a.policy,
            retries: b.retries ?? a.retries,
            interval: b.interval ?? a.interval,
            methods: b.methods ?? a.methods));
    if (o == null ||
        o.enabled == false ||
        ctx.get(sourceProbe) == true ||
        !o.methods!.contains(ctx.request.method)) {
      return next();
    }
    final cancel = ctx.cancelled();
    if (cancel != null) return cancel;
    final key = ctx.key();
    if (key == null) return next();
    final table = tables[o.policy]!;
    final existing = table[key];
    final entry = existing ??
        (table[key] =
            _Shared(ctx.fail(const DiomanFailure(kind: DiomanKind.cancel))));
    final shared = o.policy == DiomanSharePolicy.start ||
        o.policy == DiomanSharePolicy.retry;
    entry.subscribers++;
    if (shared) {
      if (existing == null) {
        if (o.policy == DiomanSharePolicy.retry) ctx.markReplayable();
        unawaited(run(ctx, next, o, table, key, entry));
      }
    } else {
      unawaited(compete(ctx, next, o, table, key, entry));
    }
    try {
      final r =
          await ctx.wait(entry.result.future, timeout: timeout(ctx.request));
      return ctx.cancelled() ??
          r ??
          ctx.fail(const DiomanFailure(kind: DiomanKind.timeout));
    } finally {
      if (--entry.subscribers == 0) {
        detach(table, key, entry);
        if (shared && !entry.result.isCompleted) entry.token.cancel();
      }
    }
  }

  /// 清空所有策略表并取消共享底层传输。
  @override
  void dispose() {
    for (final table in tables.values) {
      for (final entry in table.values) {
        entry.token.cancel();
        if (!entry.result.isCompleted) entry.result.complete(entry.cancelled);
      }
      table.clear();
    }
  }
}

import 'dart:async';
import 'package:dioman/pipeline.dart';

/// 可确定推进的时钟，不使用真实睡眠。
class FakeClock implements DiomanClock {
  /// 创建起点为 Unix epoch 的时钟。
  FakeClock();

  /// 当前虚拟时间。
  DateTime _now = DateTime.fromMillisecondsSinceEpoch(0);

  /// 所有尚未触发的计时器。
  final _timers = <_FakeTimer>[];

  /// 未清理计时器数。
  int get pendingTimers => _timers.where((t) => t.isActive).length;

  /// 返回虚拟时间。
  @override
  DateTime now() => _now;

  /// 登记单次计时器。
  @override
  Timer createTimer(Duration duration, void Function() callback) {
    final timer = _FakeTimer(_now.add(duration), callback);
    _timers.add(timer);
    return timer;
  }

  /// 推进 [duration]，依到期顺序执行回调，回调可创建新计时器。
  void elapse(Duration duration) {
    final end = _now.add(duration);
    while (true) {
      final due = _timers
          .where((t) => t.isActive && !t.deadline.isAfter(end))
          .toList()
        ..sort((a, b) => a.deadline.compareTo(b.deadline));
      if (due.isEmpty) break;
      final timer = due.first;
      _now = timer.deadline;
      timer.fire();
    }
    _timers.removeWhere((t) => !t.isActive);
    _now = end;
  }
}

/// 单次虚拟计时器。
class _FakeTimer implements Timer {
  /// 保存截止时间及回调。
  _FakeTimer(this.deadline, this.callback);

  /// 截止时间。
  final DateTime deadline;

  /// 用户回调。
  final void Function() callback;

  /// 是否仍登记。
  bool _active = true;

  /// 触发次数。
  int _tick = 0;

  /// 当前活动状态。
  @override
  bool get isActive => _active;

  /// 已触发次数。
  @override
  int get tick => _tick;

  /// 取消后不再触发。
  @override
  void cancel() => _active = false;

  /// 触发一次并关闭。
  void fire() {
    _active = false;
    _tick++;
    callback();
  }
}
