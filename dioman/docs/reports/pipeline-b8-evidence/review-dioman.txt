  void set<T>(StateKey<T> key, T value) => (state.values ??= {})[key] = value;

  bool canReplay() {

    if (value == null ||

        value is String ||

    if (value is List) return value.every(_canReplay);

  /// 已取消立即返回；迟到的值或异常被消费且不会改变结局。
  Future<T?> wait<T>(Future<T> pending, {Duration? timeout}) {

    void clean() {
      timer?.cancel();
      localLink?.dispose();
      state.listeners.remove(cancel);

    void finish(T? value) {
      if (result.isCompleted) return;
      clean();
      result.complete(value);
    }

    cancel = () => finish(null);
    // 即使已经取消，也消费输入的迟到异常。
    pending.then((value) => finish(value),
        onError: (Object error, StackTrace stack) {
      if (result.isCompleted) return;
      clean();
      result.completeError(error, stack);
    });
    if (isCancelled) {
      finish(null);
      return result.future;
    }
    state.listeners[cancel] = cancelToken;
    if (!identical(cancelToken, state.token)) {
      localLink = DiomanCancelLink(cancelToken, (_) => cancel());
    }
    if (timeout != null) timer = clock.createTimer(timeout, cancel);
    return result.future;
  }

/// 可解除目标的 Dio 取消桥；Dio Future 自身没有退订接口。
class DiomanCancelLink {
  /// 将 [source] 联动到 [target]，已取消时同步传递。
  DiomanCancelLink(CancelToken source, void Function(Object?) target)
      : _target = target {
    if (source.isCancelled) {
      target(source.cancelError);
    } else {
      source.whenCancel.then(_deliver);
    }
  }

  /// 可清除的订阅目标，清理后不再引用请求状态。
  void Function(Object?)? _target;

  /// 转发用户取消。
  void _deliver(Object? reason) => _target?.call(reason);

  /// 解除逻辑订阅；迟到的取消不再触及请求。
  void dispose() => _target = null;
}

/// 一个逻辑请求的共享状态和取消订阅登记。
class DiomanContextState {
  /// 创建状态并监听内核令牌；测试也可独立构造上下文。
  DiomanContextState(this.token, {void Function()? onDispose})
      : _onDispose = onDispose {
    _link = DiomanCancelLink(token, (_) => _notify());
  }

  /// 最后一个传输释放时通知客户端摘除取消登记。
  final void Function()? _onDispose;

  /// 内核逻辑取消令牌。
  final CancelToken token;

  /// Dio 到内核订阅表的桥。
  late final DiomanCancelLink _link;

  /// 惰性分配的类型化值。
  Map<Object, Object?>? values;

  /// 按请求身份缓存 key。
  final keys = Map<DiomanRequest, String?>.identity();

  /// 等待取消的订阅表。
  final listeners = <void Function(), CancelToken>{};

  /// 是否需要克隆发送体。
  bool replayable = false;

  /// 已发送的不可重放载荷，按身份记录，防止补丁视图绕过保护。
  final consumedBodies = Set<Object>.identity();

  /// 脱离首发调用方但仍在执行的共享传输数。
  int _retained = 0;

  /// 逻辑调用方是否已完成。
  bool _released = false;

  /// 保持共享传输使用的状态；每次调用必须对应一次 release。
  void retain() => _retained++;

  /// 共享传输完成后释放；最后一个持有者负责最终清理。
  void release() {
    _retained--;
    if (_released && _retained == 0) dispose();
  }

  /// 通知副本，允许回调在执行中退订。
  void _notify() {
    for (final entry in listeners.entries.toList()) {
      if (identical(entry.value, token)) entry.key();
    }
  }

  /// 同步取消内核请求并通知等待者。
  void cancel([Object? reason]) {
    token.cancel(reason);
    _notify();
  }

  /// 释放桥并结束剩余等待。
  void dispose() {
    _released = true;
    if (_retained > 0) return;
    _link.dispose();
    for (final listener in listeners.keys.toList()) {
      listener();
    }
    listeners.clear();
    _onDispose?.call();
  }
}

/// 每客户端按策略隔离的活动表。
class _Share<R extends DiomanResponse<dynamic>> extends DiomanPluginInstance<R>
    implements DiomanShareApi {
  /// 保存配置。
  _Share(this.base);

  /// 预合并配置。
  final DiomanShareOptions base;

  /// 每种策略一张表。
  final tables = {
    for (final p in DiomanSharePolicy.values) p: <String, _Shared>{}
  };

  /// 仅移除仍属于本轮的条目。
  void detach(Map<String, _Shared> table, String key, _Shared entry) {
    if (identical(table[key], entry)) table.remove(key);
  }

  /// 结果发布前先摘表，外层 retry 可立即重新入表。
  void finish(Map<String, _Shared> table, String key, _Shared entry,
      DiomanResponse<dynamic> r) {
    if (entry.result.isCompleted) return;
    detach(table, key, entry);
    entry.result.complete(r);
  }

  /// 订阅者整体上限为连接与接收之和；有请求体时再加发送阶段。
  /// 任一参与阶段为 null 或非正数时不限时，不模拟 Dio 的阶段进度。
  Duration? timeout(DiomanRequest request) {
    var total = Duration.zero;
    for (final stage in [
      request.connectTimeout,
      request.receiveTimeout,
      if (request.data != null) request.sendTimeout
    ]) {
      if (stage == null || stage <= Duration.zero) return null;
      total += stage;
    }
    return total;
  }

  /// 首发传输脱离首发调用方，最后一个订阅者离开才取消。
  Future<void> run(
      DiomanContext<R> ctx,
      DiomanNext<R> next,
      DiomanShareOptions o,
      Map<String, _Shared> table,
      String key,
      _Shared entry) async {
    final shared = ctx.derive(entry.transportPatch);
    ctx.state.retain();
    try {
      var r = await next(entry.transportPatch);
      if (o.policy == DiomanSharePolicy.retry) {
        for (var i = 0;
            i < o.retries! &&
                !r.ok &&
                r.kind != DiomanKind.internal &&
                shared.canReplay() &&
                !shared.isCancelled;
            i++) {
          if (o.interval! > Duration.zero) {
            await shared.wait(Completer<void>().future, timeout: o.interval);
          }
          if (shared.isCancelled) break;
          r = await next(entry.transportPatch);
        }
      }
      finish(table, key, entry, r);
    } catch (error, stack) {
      finish(
          table,
          key,
          entry,
          ctx.fail(DiomanFailure(
              kind: DiomanKind.internal, error: error, errorStack: stack)));
    } finally {
      ctx.state.release();
    }
  }

  /// end/race 的底层请求各自独立，订阅者离开不会取消其它传输。
  Future<void> compete(
      DiomanContext<R> ctx,
      DiomanNext<R> next,
      DiomanShareOptions o,
      Map<String, _Shared> table,
      String key,
      _Shared entry,
      int sequence) async {
    ctx.state.retain();
    try {
      final r = await next();
      if (ctx.cancelled() != null) {
        entry.candidates.remove(sequence);
      } else if (entry.candidates.containsKey(sequence)) {
        entry.candidates[sequence] = r;
        entry.completed.add(sequence);
      }
      settle(o, table, key, entry);
    } finally {
      ctx.state.release();
    }
  }

  /// 撤回后重新选择最新有效请求或最后完成的有效失败。
  void settle(DiomanShareOptions o, Map<String, _Shared> table, String key,
      _Shared entry) {
    if (entry.candidates.isEmpty) return;
    if (o.policy == DiomanSharePolicy.end) {
      final r = entry.candidates[entry.candidates.keys.last];
      if (r != null) finish(table, key, entry, r);
    } else {
      final done = entry.completed.where(entry.candidates.containsKey).toList();
      for (final id in done) {
        final r = entry.candidates[id]!;
        if (r.ok) {
          finish(table, key, entry, r);
          return;
        }
      }
      if (done.length == entry.candidates.length) {
        finish(table, key, entry, entry.candidates[done.last]!);
      }
    }
  }

  /// 解析 optIn 后订阅；wait 自行清理取消监听和计时器。
  @override
  Future<DiomanResponse<dynamic>> handle(
      DiomanContext<R> ctx, DiomanNext<R> next) async {
    final o = resolveOption<DiomanShareOptions>(
        ctx.request.call.options[DiomanShare.option],
        base: base,
        optIn: true,
        shorthand: (v) =>
            v is DiomanSharePolicy ? DiomanShareOptions(policy: v) : null,
        merge: (a, b) => DiomanShareOptions(
            enabled: b.enabled,
            policy: b.policy ?? a.policy,
            retries: b.retries ?? a.retries,
            interval: b.interval ?? a.interval,
            methods: b.methods ?? a.methods));
    if (o == null ||
        o.enabled == false ||
        ctx.get(sourceProbe) == true ||
        !o.methods!.contains(ctx.request.method)) {
      return next();
    }
    final cancel = ctx.cancelled();
    if (cancel != null) return cancel;
    final key = ctx.key();
    if (key == null) return next();
    final table = tables[o.policy]!;
    final existing = table[key];
    final entry = existing ??
        (table[key] =
            _Shared(ctx.fail(const DiomanFailure(kind: DiomanKind.cancel))));
    final shared = o.policy == DiomanSharePolicy.start ||
        o.policy == DiomanSharePolicy.retry;
    entry.subscribers++;
    final sequence = ++entry.sequence;
    if (!shared) entry.candidates[sequence] = null;
    if (shared) {
      if (existing == null) {
        if (o.policy == DiomanSharePolicy.retry) ctx.markReplayable();
        unawaited(run(ctx, next, o, table, key, entry));
      }
    } else {
      unawaited(compete(ctx, next, o, table, key, entry, sequence));
    }
    try {
      final r =
          await ctx.wait(entry.result.future, timeout: timeout(ctx.request));
      if (!shared && (ctx.cancelled() != null || r == null)) {
        entry.candidates.remove(sequence);
        settle(o, table, key, entry);
      }
      return ctx.cancelled() ??
          r ??
          ctx.fail(const DiomanFailure(kind: DiomanKind.timeout));
    } finally {
      if (--entry.subscribers == 0) {
        detach(table, key, entry);
        if (shared && !entry.result.isCompleted) entry.token.cancel();
      }
    }
  }

  /// 清空所有策略表并取消共享底层传输。
  @override
  void dispose() {
    for (final table in tables.values) {
      for (final entry in table.values) {
        entry.token.cancel();
        if (!entry.result.isCompleted) entry.result.complete(entry.cancelled);
      }
      table.clear();
    }
  }

    final explicit = override is String ? override : o?.key;
    if (o?.builder != null) return _scopedKey(o!.builder!(request), scope);
    if (explicit != null && explicit.isNotEmpty) {
      return _scopedKey(explicit, scope);
    }
    if (options.builder != null) {
      return _scopedKey(options.builder!(request), scope);
    }
    final ignores = _merge(options.ignores ?? const [], o?.ignores);
    return (o?.fastMode ?? options.fastMode ?? false)
        ? _scopedKey(_fastKey(request), scope)
        : _deepKey(request, ignores, scope);
  }

  /// 为快速键或自定义键增加独立作用域字段，空作用域保持原键。
  static String _scopedKey(String key, String? scope) =>
      scope == null || scope.isEmpty ? key : jsonEncode([key, scope]);

  /// Merges [override] into [base] (union), or returns [base] unchanged when
  /// [override] is `null`.
  ///
  /// 把[override]与[base]取并集合并；[override]为`null`时原样返回[base]。
  static List<String> _merge(List<String> base, List<String>? override) =>
      override == null ? base : {...base, ...override}.toList();

  /// 使用方法、来源和路径生成快速键。
  String _fastKey(RequestOptions o) =>
      '${o.method.toUpperCase()}:${o.uri.hasAuthority ? o.uri.origin : ''}${o.uri.path}';

  /// 分字段编码规范化查询及请求体，保留非结构化对象的身份。
  String? _deepKey(RequestOptions o, List<String> ignoreKeys, String? scope) {
    final uri = o.uri;
    final params = <String, dynamic>{
      for (final e in uri.queryParametersAll.entries)
        if (!ignoreKeys.contains(e.key))
          e.key: e.value.length == 1 ? e.value.single : e.value,
    };
    final data = o.data;
    Object? body;
    if (data is Map) {
      final filtered = <String, dynamic>{
        for (final e in data.entries)
          if (!ignoreKeys.contains('${e.key}')) '${e.key}': e.value,
      };
      if (filtered.isNotEmpty) {
        final encoded = _encode(filtered);
        if (encoded == null) return null;
        body = {'map': encoded};
      }
    } else if (data is String) {
      if (data.isNotEmpty) body = {'string': data};
    } else if (data is List<int>) {
      if (data.length > maxBodyBytes) return null;
      var hash = 0x811c9dc5;
      for (final byte in data) {
        hash = ((hash ^ byte) * 0x01000193) & 0xffffffff;
      }
      body = {
        'bytes': [data.length, hash]
      };
    } else if (data is List) {
      final encoded = _encode(data);
      if (encoded == null) return null;
      body = {'list': encoded};
    } else if (data != null) {
      return null;
    }
    final query = _encode(params);
    if (query == null) return null;
    return jsonEncode([
      _fastKey(o),
      query,
      body,
      if (scope != null && scope.isNotEmpty) scope,
    ]);
  }

  /// 递归排序 Map 字段并保留 List 顺序后编码，失败则禁用自动键。
  static String? _encode(Object v) {
    try {
      return jsonEncode(_normalize(v));
    } catch (_) {
      return null;
    }
  }

  /// 返回递归规范化的副本：Map键按字符串比较排序，List保序，其他值原样保留。
  static Object? _normalize(Object? v) {
    if (v is Map) {
      final entries = v.entries.toList()
        ..sort((a, b) => '${a.key}'.compareTo('${b.key}'));
      return Map.fromEntries(
        entries.map((e) => MapEntry(e.key, _normalize(e.value))),
      );
    }
    if (v is List) return v.map(_normalize).toList();
    return v;
  }
}

/// 全部 HTTP 状态交给工厂，不让 Dio 构造 badResponse。
bool _acceptStatus(int? _) => true;

/// 递归克隆 JSON 请求体；FormData 使用 Dio 的 clone，其他类型保留身份。
Object? _clone(Object? value, DiomanLogger log) {
  if (value is Map) return value.map((k, v) => MapEntry(k, _clone(v, log)));
  if (value is List) return value.map((v) => _clone(v, log)).toList();
  if (value is FormData) return value.clone();
  if (value != null && value is! num && value is! bool && value is! String) {
    log('请求体无法克隆，回退原引用');
  }
  return value;
}

/// 执行一次发送；传输构造、Dio 及工厂异常都在本边界归一化。
Future<DiomanResponse<dynamic>>
    diomanTransport<R extends DiomanResponse<dynamic>>(
        Dio dio, DiomanContext<R> ctx) async {
  try {
    final cancelled = ctx.cancelled();
    if (cancelled != null) return cancelled;
    if (!ctx.canReplay() && !ctx.state.consumedBodies.add(ctx.request.data!)) {
      throw StateError('插件错误：不可重放请求体被重复发送');
    }
    final options = ctx.request.toRequestOptions()
      ..validateStatus = _acceptStatus
      ..cancelToken = ctx.cancelToken;
    if (ctx.replayable) options.data = _clone(options.data, ctx.log);
    final raw = await dio.fetch<dynamic>(options);
    return ctx.factory
        .normalize(ctx.request, raw: raw, cancelled: ctx.isCancelled);
  } catch (e, s) {
    return ctx.factory
        .normalize(ctx.request, error: e, stack: s, cancelled: ctx.isCancelled);
  }
}

/// 鉴权失败后的处理动作。
enum DiomanAuthFailureAction {
  /// 加入或发起单飞刷新。
  refresh,

  /// 用当前凭证直接重放。
  replay,

  /// 通知访问拒绝。
  deny,

  /// 结束失败请求所属会话。
  expired,

  /// 原样返回，与鉴权无关。
  others,
}

/// 通知回调，异常隔离；接收存储与失败结果。
typedef DiomanAuthNotice = FutureOr<void> Function(
    DiomanTokenManager tokens, DiomanResponse<dynamic> result);

/// 刷新回调自行调用 setTokens；同步与异步异常均算刷新失败。
typedef DiomanAuthRefresh = FutureOr<void> Function(
    DiomanTokenManager tokens, DiomanResponse<dynamic> result);

/// 判定回调，异常按程序错误落定。
typedef DiomanAuthFailure = FutureOr<DiomanAuthFailureAction> Function(
    DiomanResponse<dynamic> result, DiomanTokenManager tokens);

typedef DiomanAuthReady = FutureOr<RequestPatch> Function(
    DiomanTokenManager tokens, DiomanRequest request);

/// 默认按逻辑状态判定；[injectedToken] 是最后一次调用 ready 前的凭证快照。
/// 例如旧请求携带 a、当前为 b 时返回 replay，保留新会话。
DiomanAuthFailureAction defaultAuthFailure(DiomanResponse<dynamic> result,
    DiomanTokenManager tokens, String? injectedToken) {
  if (result.status == 403) return DiomanAuthFailureAction.deny;
  if (result.status != 401) return DiomanAuthFailureAction.others;
  final current = tokens.accessToken;
  if (current == null || current.isEmpty) {
    return DiomanAuthFailureAction.expired;
  }
  return injectedToken != current
      ? DiomanAuthFailureAction.replay
      : DiomanAuthFailureAction.refresh;
}

    } catch (e, s) {
      ctx.client.logError('[${DiomanAuth.pluginName}] 鉴权通知失败: $e\n$s');
    }
  }

  /// 只结束失败请求所属会话，旧请求不能清新凭证。
  Future<void> _expire(DiomanContext<R> ctx, DiomanAuthOptions o,
      DiomanResponse<dynamic> r, String? session) async {
    final current = _tokens.accessToken;
    if (_hasToken && current != session) return;
    if (!_hasToken && session == _expiredSession) return;
    _expiredSession = session;
    _tokens.clear();
    await _notice(ctx, o.onAccessExpired!, r);
  }

  /// 启动强制异步刷新；窗口发布先于用户回调，同步抛错也必释放。
  Future<bool> _refresh(DiomanContext<R> ctx, DiomanAuthOptions o,
      DiomanResponse<dynamic> r, String? session) {
    return _refreshing ??= (() async {
      try {
        final done = Completer<void>();
        final timer = ctx.clock.createTimer(o.refreshTimeout!, () {
          if (!done.isCompleted) done.completeError(TimeoutException('刷新超时'));
        });
        Future<void>(() => config.onRefresh(_tokens, r)).then((_) {
          if (!done.isCompleted) done.complete();
        }, onError: (Object e, StackTrace s) {
          if (!done.isCompleted) done.completeError(e, s);
        });
        try {
          await done.future;
        } finally {
          timer.cancel();
        }
        return true;
      } catch (e, s) {
        ctx.log('[${DiomanAuth.pluginName}] 刷新失败: $e\n$s');
        if (_tokens.accessToken != session) return true;
        await _expire(ctx, o, r, session);
        return false;
      } finally {
        _refreshing = null;
      }
    })();
  }

  /// 失败载体统一由工厂创建。
  DiomanResponse<dynamic> _fail(DiomanContext<R> ctx, String reason) =>
      ctx.fail(DiomanFailure(kind: DiomanKind.auth, reason: reason));

  /// 加入刷新后立刻重查本调用的取消，不中止共享刷新。
  Future<DiomanResponse<dynamic>?> _join(
      DiomanContext<R> ctx, Future<bool> pending) async {
    final success = await ctx.wait(pending);
    final cancelled = ctx.cancelled();
    return cancelled ?? (success == true ? null : _fail(ctx, _expired));
  }

  /// 每次尝试入口求值保护规则，同一尝试内至多重放一次。
  @override
  Future<DiomanResponse<dynamic>> handle(
      DiomanContext<R> ctx, DiomanNext<R> next) async {
    final o = resolveOption<DiomanAuthOptions>(
        ctx.request.call.options[DiomanAuth.option],
        base: base,
        merge: _merge);
    if (o == null ||
        o.enabled == false ||
        (o.enabled != true && o.isProtected?.call(ctx.request) == false)) {
      return next();
    }
    ctx.markReplayable();
    final pending = _refreshing;
    if (pending != null) {
      final failure = await _join(ctx, pending);
      final cancelled = ctx.cancelled();
      if (cancelled != null) return cancelled;
      if (failure != null) return failure;
    }
    var replayed = false;
    var proactive = false;
    while (true) {
      // 重放前必须重查凭证，刷新成功也可能已被使用者登出。
      if (!_hasToken) {
        final r = _fail(ctx, 'unauthenticated');
        await ctx.wait(_notice(ctx, o.onAccessDenied ?? o.onAccessExpired!, r));
        return ctx.cancelled() ?? r;
      }
      // 登录（包括拿到同名凭证）之后允许新一轮过期通知。
      _expiredSession = null;
      final at = o.expiresAt?.call(_tokens.accessToken!);
      if (!proactive &&
          at != null &&
          !at.isAfter(ctx.clock.now().add(o.refreshLeeway!))) {
        proactive = true;
        final failure = await _join(
            ctx, _refresh(ctx, o, _fail(ctx, _expired), _tokens.accessToken));
        final cancelled = ctx.cancelled();
        if (cancelled != null) return cancelled;
        if (failure != null) return failure;
        continue;
      }
      RequestPatch? patch;
      String? token;
      for (var i = 0; i < DiomanAuth.readyStableAttempts; i++) {
        token = _tokens.accessToken;
        patch = o.ready == null
            ? RequestPatch(headers: {o.headerKey!: o.buildHeader!(token!)})
            : await ctx.wait(Future<RequestPatch>.sync(
                () => o.ready!(_tokens, ctx.request)));
        final cancelled = ctx.cancelled();
        if (cancelled != null) return cancelled;
        final current = _tokens.accessToken;
        if (current == token) break;
        if (!_hasToken) break;
      }
      if (!_hasToken) continue;
      ctx.set(DiomanAuth.injectedToken, token);
      final r = await next(patch);
      final cancelled = ctx.cancelled();
      if (r.kind == DiomanKind.internal) return r;
      if (cancelled != null) return cancelled;
      if (r.ok || r.kind == DiomanKind.cancel) return r;
      final action = o.onFailure == null
          ? defaultAuthFailure(
              r, _tokens, ctx.get<String?>(DiomanAuth.injectedToken))
          : await ctx.wait(Future<DiomanAuthFailureAction>.sync(
              () => o.onFailure!(r, _tokens)));
      final cancelledAfterFailure = ctx.cancelled();
      if (cancelledAfterFailure != null) return cancelledAfterFailure;
      switch (action!) {
        case DiomanAuthFailureAction.refresh:
        case DiomanAuthFailureAction.replay:
          if (!ctx.canReplay()) return r;
          if (replayed) {
            await ctx.wait(_expire(ctx, o, r, token));
            return ctx.cancelled() ?? _fail(ctx, _expired);
          }
          if (action == DiomanAuthFailureAction.refresh) {
            final failure = await _join(ctx, _refresh(ctx, o, r, token));
            final cancelled = ctx.cancelled();
            if (cancelled != null) return cancelled;
            if (failure != null) return failure;
          }
          replayed = true;
        case DiomanAuthFailureAction.expired:
          await ctx.wait(_expire(ctx, o, r, token));
          return ctx.cancelled() ?? _fail(ctx, _expired);
        case DiomanAuthFailureAction.deny:
          final denied = _fail(ctx, 'denied');
          await ctx.wait(
              _notice(ctx, o.onAccessDenied ?? o.onAccessExpired!, denied));
          return ctx.cancelled() ?? denied;
        case DiomanAuthFailureAction.others:
          return r;
      }
    }
  }

/// 重发内层管线，每次尝试重新经过熔断与鉴权。
class DiomanRetry<R extends DiomanResponse<dynamic>> extends DiomanPlugin<R> {
  /// 创建插件；请求级使用 option 键覆盖，状态由 setup 隔离。
  /// 例如 `DiomanRetry(enabled: false)` 可关闭安装。
  const DiomanRetry(
      {this.enabled = true,
      this.max = 0,
      this.methods = defaultMethods,
      this.statusCodes = defaultStatusCodes,
      this.shouldRetry,
      this.delay = defaultDelay,
      this.jitter,
      this.delayMax,
      this.respectRetryAfter = true,
      this.afterStatusCodes = defaultAfterStatusCodes,
      this.retryAfterMax = const Duration(seconds: 60),
      this.onRetry});

  /// 安装或请求级开关。
  @override
  final bool enabled;

  /// 最多重试次数；非正数只发送一次。
  final int max;

  /// 允许重试的方法，忽略大小写，优先于自定义规则。
  final List<String> methods;

  /// 触发重试的逻辑状态表。
  final List<int> statusCodes;

  /// 自定义判定；null 回退默认规则，也接收成功响应。
  final DiomanShouldRetry? shouldRetry;

  /// 退避计算，参数为当前重试序号、上限与结果。
  final DiomanRetryDelay delay;

  /// true 随机抖动，或 DiomanJitter 函数。
  final Object? jitter;

  /// 抖动后的退避上限；null 不封顶。
  final Duration? delayMax;

  /// 优先遵守 Retry-After。
  final bool respectRetryAfter;

  /// 接受 Retry-After 的逻辑状态。
  final List<int> afterStatusCodes;

  /// 服务器等待时间上限；不受 delayMax 限制。
  final Duration retryAfterMax;

  /// 重试通知；抛错隔离并记日志。
  final void Function(int attempt)? onRetry;

  /// 默认幂等方法白名单。
  static const defaultMethods = [
    'GET',
    'PUT',
    'HEAD',
    'DELETE',
    'OPTIONS',
    'TRACE'
  ];

  /// 默认逻辑失败状态。
  static const defaultStatusCodes = [408, 429, 500, 502, 503, 504];

  /// 默认接受服务器退避的状态。
  static const defaultAfterStatusCodes = [413, 429, 503];

  /// 默认固定三秒退避。
  static Duration defaultDelay(

  /// 稳定插件名称。

  @override

  @override

  @override

/// 每客户端的退避与随机源。
class _Retry<R extends DiomanResponse<dynamic>>
    extends DiomanPluginInstance<R> {
  /// 保存预合并配置。
  _Retry(this.config, this.base);

  /// 插件配置。
  final DiomanRetry<R> config;

  /// 预合并配置。
  final DiomanRetryOptions base;

  /// 随机抖动源。
  final _random = Random();

  /// 不接受重试的失败类别。
  static const _veto = {
    DiomanKind.cancel,
    DiomanKind.auth,
    DiomanKind.internal,
    DiomanKind.parse,
    DiomanKind.breaker,
    DiomanKind.offline
  };

  /// RFC1123 日期模式。
  static final _httpDate =
      RegExp(r'^\w{3}, (\d{2}) (\w{3}) (\d{4}) (\d{2}):(\d{2}):(\d{2}) GMT$');

  /// HTTP 日期月份。
  static const _months = [
    'Jan',
    'Feb',
    'Mar',
    'Apr',
    'May',
    'Jun',
    'Jul',
    'Aug',
    'Sep',
    'Oct',
    'Nov',
    'Dec'
  ];

  /// 统一裁剪负数与可选上限。
  Duration _cap(Duration value, Duration? cap) => value.isNegative
      ? Duration.zero
      : cap != null && value > cap
          ? cap
          : value;

  /// 严格按类别、自定义、网络、逻辑状态的顺序判定。
  bool _retry(DiomanRetryOptions o, DiomanResponse<dynamic> r, int attempt) {
    if (_veto.contains(r.kind)) return false;
    final custom = o.shouldRetry?.call(r, attempt);
    if (custom != null) return custom;
    return r.kind == DiomanKind.network ||
        r.kind == DiomanKind.timeout ||
        o.statusCodes!.contains(r.status);
  }

  /// 解析服务器秒数或 HTTP 日期；无效值回退普通退避。
  Duration? _after(
      DiomanContext<R> ctx, DiomanRetryOptions o, DiomanResponse<dynamic> r) {
    if (!o.respectRetryAfter! || !o.afterStatusCodes!.contains(r.status)) {
      return null;
    }
    final raw = r.raw?.headers.value('retry-after')?.trim();
    if (raw == null) return null;
    final seconds = raw.isEmpty ? 0 : num.tryParse(raw);
    if (seconds != null && seconds.isFinite) {
      return _cap(Duration(milliseconds: (seconds * 1000).round()), null);
    }
    final m = _httpDate.firstMatch(raw);
    if (m == null) return null;
    final month = _months.indexOf(m[2]!);
    if (month < 0) return null;
    final date = DateTime.utc(int.parse(m[3]!), month + 1, int.parse(m[1]!),
        int.parse(m[4]!), int.parse(m[5]!), int.parse(m[6]!));
    return _cap(date.difference(ctx.clock.now().toUtc()), null);
  }

  /// 服务器退避优先；普通退避施加抖动后封顶。
  Duration _delay(DiomanContext<R> ctx, DiomanRetryOptions o,
      DiomanResponse<dynamic> r, int attempt) {
    final after = _after(ctx, o, r);
    if (after != null) return after;
    final raw = o.delay!(attempt, o.max!, r);
    var value = raw;
    final jitter = o.jitter;
    if (jitter == true) {
      value = Duration(
          microseconds: (raw.inMicroseconds * _random.nextDouble()).round());
    } else if (jitter is DiomanJitter) {
      final changed = jitter(raw);
      if (!changed.isNegative) value = changed;
    }
    return _cap(value, o.delayMax);
  }

  /// 从相同基准重发，只恢复本层实际放行的请求。
  @override
  Future<DiomanResponse<dynamic>> handle(
      DiomanContext<R> ctx, DiomanNext<R> next) async {
    final o = resolveOption<DiomanRetryOptions>(
        ctx.request.call.options[DiomanRetry.option],
        base: base,
        merge: _merge,
        shorthand: (v) => v is int ? DiomanRetryOptions(max: v) : null);
    if (o == null || o.enabled == false) return next();
    ctx.markReplayable();
    if (!o.methods!
        .map((m) => m.toUpperCase())
        .contains(ctx.request.method.toUpperCase())) {
      return next();
    }
    var r = await next();
    var cancelled = ctx.cancelled();
    if (r.kind == DiomanKind.internal) return r;
    if (cancelled != null) return cancelled;
    for (var attempt = 1;
        attempt <= o.max! && _retry(o, r, attempt);

      if (!ctx.canReplay()) return r;
      final after = _after(ctx, o, r);
      if (after != null && after > o.retryAfterMax!) return r;
      final duration = _delay(ctx, o, r, attempt);
      // ctx.wait 自己拥有计时器，取消时即释放，无额外退避资源。
      await ctx.wait<void>(Completer<void>().future, timeout: duration);
      cancelled = ctx.cancelled();
      if (cancelled != null) return cancelled;
      try {
        o.onRetry?.call(attempt);
      } catch (e, s) {
        ctx.client.logError('[${DiomanRetry.pluginName}] 重试通知失败: $e\n$s');
      }
      cancelled = ctx.cancelled();
      if (cancelled != null) return cancelled;
      r = await next();
      cancelled = ctx.cancelled();
      if (r.kind == DiomanKind.internal) return r;
      if (cancelled != null) return cancelled;
    }
    return r;
  }

/// 有效 Dio 配置的不可变请求视图；入口合并一次后构造。
class DiomanRequest {
  /// 从 [options] 拷贝有效配置；[call] 保存调用选项，例如 `DiomanRequest(ro)`。
  DiomanRequest(RequestOptions options, {DiomanCall call = const DiomanCall()})
      : _options = options.copyWith(),
        data = options.data,
        headers = Map.unmodifiable(options.headers),
        query = Map.unmodifiable(options.queryParameters),
        extra = Map.unmodifiable(options.extra),
        call = DiomanCall(
            key: call.key,
            envelope: call.envelope,
            silent: call.silent,
            options: Map.unmodifiable(call.options));

  /// 完整有效配置的私有副本，永不交给 Dio 修改。
  final RequestOptions _options;

  /// 请求体引用；不透明载荷由传输按需克隆，插件通过补丁替换。
  final Object? data;

  /// 不可修改的请求头。
  final Map<String, dynamic> headers;

  /// 不可修改的查询参数。
  final Map<String, dynamic> query;

  /// 不可修改的用户扩展字段。
  final Map<String, dynamic> extra;

  /// 调用配置。
  final DiomanCall call;

  /// 有效路径。
  String get path => _options.path;

  /// 有效方法。
  String get method => _options.method;

  /// 有效基础地址。
  String get baseUrl => _options.baseUrl;

  /// 有效 URI。
  Uri get uri => toRequestOptions().uri;

  /// 连接超时。
  Duration? get connectTimeout => _options.connectTimeout;

  /// 发送超时。
  Duration? get sendTimeout => _options.sendTimeout;

  /// 接收超时。
  Duration? get receiveTimeout => _options.receiveTimeout;

  /// 当前传输令牌；未指定时内核使用逻辑请求令牌。
  CancelToken? get cancelToken => _options.cancelToken;

  /// 通知静默标记。
  bool get silent => call.silent;

  /// 返回可独立修改的有效 Dio 配置，例如 `dio.fetch(r.toRequestOptions())`。
  RequestOptions toRequestOptions() => _options.copyWith(
      headers: Map.from(headers),
      queryParameters: Map.from(query),
      extra: Map.from(extra))
    ..data = data;

  /// 合并 [patch] 并返回独立视图；不修改基准请求。
  DiomanRequest patch(RequestPatch patch) {
    final o = toRequestOptions();
    final h = LinkedHashMap<String, dynamic>.from(o.headers);
    for (final key in patch.removeHeaders) {
      h.removeWhere((k, _) => k.toLowerCase() == key.toLowerCase());
    }
    for (final e in patch.headers.entries) {
      h.removeWhere((k, _) => k.toLowerCase() == e.key.toLowerCase());
      h[e.key] = e.value;
    }
    o.headers = h;
    o.queryParameters.removeWhere((k, _) => patch.removeQuery.contains(k));
    o.queryParameters.addAll(patch.query);
    o.extra.addAll(patch.extra);
    if (!identical(patch.data, _absent)) o.data = patch.data;
    if (patch.path != null) o.path = patch.path!;
    if (patch.method != null) o.method = patch.method!;
    if (!identical(patch.connectTimeout, _absent)) {
      o.connectTimeout = patch.connectTimeout as Duration?;
    }
    if (!identical(patch.sendTimeout, _absent)) {
      o.sendTimeout = patch.sendTimeout as Duration?;
    }
    if (!identical(patch.receiveTimeout, _absent)) {
      o.receiveTimeout = patch.receiveTimeout as Duration?;
    }
    if (!identical(patch.cancelToken, _absent)) {
      o.cancelToken = patch.cancelToken as CancelToken?;
    }
    return DiomanRequest(o, call: call);
  }
}
