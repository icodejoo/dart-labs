/// 关闭 debug 时复用的空日志器。
void _noLog(String _) {}

/// 以 Dio 为引擎的洋葱管线客户端；旧入口与本库完全独立。
class Dioman<R extends DiomanResponse<dynamic>> {
  /// 创建客户端，例如 `Dioman(dio: Dio(), debug: true)`。
  /// [response] 控制归一化，[reject] 只转换动词方法的可预期失败。
  Dioman(
      {required this.dio,
      List<DiomanPlugin<R>> plugins = const [],
      DiomanResponseFactory<R>? response,
      this.key = const DiomanKeyOptions(),
      this.reject,
      Object? debug,
      this.clock = const DiomanClock.system()})
      : response = response ?? DiomanResponseFactory<R>(),
        _debug = debug != null && debug != false,
        _logger =
            debug is DiomanLogger ? debug : (debug == true ? print : _noLog) {
    if (debug != null && debug is! bool && debug is! DiomanLogger) {
      throw ArgumentError.value(debug, 'debug');
    }
    validateDiomanPlugins(plugins);
    try {
      for (final plugin in plugins) {
        _registrations.add(DiomanRegistration(
            plugin, plugin.enabled ? plugin.setup(this) : null));
      }
      _compile();
    } catch (_) {
      for (final registration in _registrations.reversed) {
        try {
          registration.instance?.dispose();
        } catch (_) {/* 保留安装时首次异常。 */}
      }
      rethrow;
    }
  }

  /// 底层 Dio；用户拦截器在每次传输时运行。
  final Dio dio;

  /// 客户端的归一化服务。
  final DiomanResponseFactory<R> response;

  /// 客户端构键配置。
  final DiomanKeyOptions key;

  /// 可预期失败的最终异常转换器。
  final Object Function(R)? reject;

  /// 可注入时间服务。
  final DiomanClock clock;

  /// 是否记录调用点。
  final bool _debug;

  /// 用户日志器或默认输出。
  final DiomanLogger _logger;

  /// 注册顺序，供生命周期和 extend 使用。
  final _registrations = <DiomanRegistration<R>>[];

  /// 已排序的执行层快照。
  List<DiomanRegistration<R>> _layers = [];

  /// 编译后的链，在途请求捕获当前值。
  late DiomanRunner<R> _run;

  /// 活跃逻辑请求。
  final _pending = <DiomanContextState>{};

  /// 仍可取消的请求状态，包含逻辑调用已结束的共享传输。
  final _cancellable = <DiomanContextState>{};

  /// 请求序号。
  int _sequence = 0;

  /// 生命周期状态。
  bool _disposed = false;

  /// 当前活跃逻辑请求数。
  int get pending => _pending.length;

  /// 隔离日志回调异常，避免 debug 改变请求结果。
  void _log(String message) {
    try {
      _logger(message);
    } catch (_) {/* 日志通知不影响请求。 */}
  }

  /// 记录插件通知故障；[message] 交给用户日志器或默认输出，日志器异常会隔离。
  /// 例如插件捕获回调错误后调用 `client.logError('通知失败')`，不受 debug 开关抑制。
  void logError(String message) {
    if (_debug) {
      _log(message);
    } else {
      try {
        print(message);
      } catch (_) {/* 输出通知不影响请求。 */}
    }
  }

  /// 拒绝销毁后的操作。
  void _checkAlive() {
    if (_disposed) throw StateError('客户端已销毁');
  }

  /// 在配置变动时编译链，不在请求期间排序。
  void _compile() {
    _layers = sortDiomanPipeline(_registrations);
    _run = compileDiomanPipeline(_layers, (ctx) => diomanTransport(dio, ctx));
  }

  /// 添加插件；先校验再 setup，安装失败不修改当前链。
  void use(DiomanPlugin<R> plugin) {
    _checkAlive();
    validateDiomanPlugins([..._registrations.map((r) => r.plugin), plugin]);
    final instance = plugin.enabled ? plugin.setup(this) : null;
    _registrations.add(DiomanRegistration(plugin, instance));
    _compile();
  }

  /// 按 [name] 移除并释放插件；不存在返回 false，在途保留原链。
  bool eject(String name) {
    final index = _registrations.indexWhere((r) => r.plugin.name == name);
    if (index < 0) return false;
    final removed = _registrations.removeAt(index);
    _compile();
    removed.instance?.dispose();
    return true;
  }

  /// 返回实现 [A] 的第一个实例，例如 `api.plugin<CacheApi>()`。
  A? plugin<A>() {
    for (final r in _registrations) {
      if (r.instance is A) return r.instance as A;
    }
    return null;
  }

  /// 返回解析后的固定阶段和名称顺序，返回列表不可修改。
  List<(DiomanStage, String)> inspect() => List.unmodifiable(_layers.map((r) {
        final fixed = diomanBuiltinStages.entries
            .where((e) => e.value.contains(r.plugin.name));
        return (
          fixed.isEmpty ? r.plugin.stage : fixed.single.key,
          r.plugin.name
        );
      }));

  /// 中止尚未取消的请求状态（含保留的共享传输），返回本次中止数。
  int cancelAll([Object? reason]) {
    var count = 0;
    for (final state in _cancellable.toList()) {
      if (!state.token.isCancelled) {
        count++;
        state.cancel(reason);
      }
    }
    return count;
  }

  /// 克隆 BaseOptions 并重新 setup 插件，状态不共享；[plugins] 替换继承列表。
  /// [options] 在 setup 后合并非空、非默认字段，显式覆盖优先；容器按键合并。
  /// BaseOptions 不记录省略信息，不能用默认值重置已有非默认值。
  /// 例如 `api.extend(options: BaseOptions(baseUrl: 'https://child.example'))`。
  Dioman<R> extend({BaseOptions? options, List<DiomanPlugin<R>>? plugins}) {
    _checkAlive();
    final child = Dioman<R>(
        dio: Dio(cloneDiomanOptions(dio.options)),
        plugins: plugins ?? _registrations.map((r) => r.plugin).toList(),
        response: response,
        key: key,
        reject: reject,
        debug: _debug ? _logger : false,
        clock: clock);
    if (options != null) mergeDiomanOptions(child.dio.options, options);
    return child;
  }

  /// 先取消请求，再逆序释放全部实例；清理完毕后原栈抛出首个异常。
  void dispose() {
    if (_disposed) return;
    _disposed = true;
    cancelAll();
    final errors = <(Object, StackTrace)>[];
    for (final r in _registrations.reversed) {
      try {
        r.instance?.dispose();
      } catch (e, s) {
        errors.add((e, s));
      }
    }
    if (errors.isNotEmpty) {
      Error.throwWithStackTrace(errors.first.$1, errors.first.$2);
    }
  }

  /// 执行请求并返回结果；可预期失败不抛，internal 原异常原栈抛出。
  /// 例如 `await api.send<Object?>('GET', '/pets')`。
  Future<DiomanResponse<T>> send<T>(String method, String path,
      {Map<String, dynamic>? query, Object? data, DiomanCall? call}) async {
    final result =
        await _dispatch(method, path, query: query, data: data, call: call);
    if (result is DiomanResponse<T>) return result;
    return result.copy<T>();
  }

  /// 执行任意 [method] 并返回数据；可预期失败经 reject 后抛出。
  Future<T> request<T>(String method, String path,
      {Map<String, dynamic>? query, Object? data, DiomanCall? call}) async {
    final result =
        await _dispatch(method, path, query: query, data: data, call: call);
    if (!result.ok) throw (reject?.call(result as R) ?? result);
    return result.data as T;
  }

  /// 发送 GET 并返回数据；参数与 request 相同，例如 `api.get<Object?>('/pets')`。
  Future<T> get<T>(String path,
          {Map<String, dynamic>? query, Object? data, DiomanCall? call}) =>
      request<T>('GET', path, query: query, data: data, call: call);

  /// 发送 POST 并返回数据；参数与 request 相同，例如 `api.post<Object?>('/pets')`。
  Future<T> post<T>(String path,
          {Map<String, dynamic>? query, Object? data, DiomanCall? call}) =>
      request<T>('POST', path, query: query, data: data, call: call);

  /// 发送 PUT 并返回数据；参数与 request 相同，例如 `api.put<Object?>('/pets')`。
  Future<T> put<T>(String path,
          {Map<String, dynamic>? query, Object? data, DiomanCall? call}) =>
      request<T>('PUT', path, query: query, data: data, call: call);

  /// 发送 PATCH 并返回数据；参数与 request 相同，例如 `api.patch<Object?>('/pets')`。
  Future<T> patch<T>(String path,
          {Map<String, dynamic>? query, Object? data, DiomanCall? call}) =>
      request<T>('PATCH', path, query: query, data: data, call: call);

  /// 发送 DELETE 并返回数据；参数与 request 相同，例如 `api.delete<Object?>('/pets')`。
  Future<T> delete<T>(String path,
          {Map<String, dynamic>? query, Object? data, DiomanCall? call}) =>
      request<T>('DELETE', path, query: query, data: data, call: call);

  /// 发送 HEAD 并返回数据；参数与 request 相同，例如 `api.head<Object?>('/pets')`。
  Future<T> head<T>(String path,
          {Map<String, dynamic>? query, Object? data, DiomanCall? call}) =>
      request<T>('HEAD', path, query: query, data: data, call: call);

  /// 入口仅 compose 一次；登记后所有可能抛错的构造都在 finally 保护内。
  Future<DiomanResponse<dynamic>> _dispatch(String method, String path,
      {Map<String, dynamic>? query, Object? data, DiomanCall? call}) async {
    _checkAlive();
    final run = _run;
    final site = _debug ? StackTrace.current : null;
    late final DiomanContextState state;
    state = DiomanContextState(CancelToken(),
        onDispose: () => _cancellable.remove(state));
    _cancellable.add(state);
    DiomanCancelLink? userLink;
    _pending.add(state);
    try {
      final c = call ?? const DiomanCall();
      if (c.cancelToken != null) {
        userLink = DiomanCancelLink(c.cancelToken!, state.cancel);
      }
      final options = Options(
              method: method,
              headers: c.headers,
              sendTimeout: c.sendTimeout,
              receiveTimeout: c.receiveTimeout)
          .compose(dio.options, path, queryParameters: query, data: data);
      if (c.connectTimeout != null) options.connectTimeout = c.connectTimeout;
      final ctx = DiomanContextInternal.create<R>(
          id: ++_sequence,
          client: this,
          request: DiomanRequest(options, call: c),
          state: state,
          factory: response,
          keyService: DiomanKeyService(key),
          clock: clock,
          log: _log);
      var result = await run(ctx);
      if (result.kind != DiomanKind.internal) {
        result = ctx.cancelled() ?? result;
      }
      if (!result.ok && site != null && result.kind != DiomanKind.internal) {
        result = response.withCallSite(result, site);
      }
      if (result.kind == DiomanKind.internal) {
        Error.throwWithStackTrace(
            result.error ?? StateError('internal result without error'),
            result.errorStack ?? StackTrace.current);
      }
      return result;
    } finally {
      userLink?.dispose();
      state.dispose();
      _pending.remove(state);
    }
  }

/// 封装 Tokens 测试夹具。
/// 令牌存储保留最近一次变化前的值。
class Tokens implements DiomanTokenManager {
  /// 完整鉴权头。
  String? access;

  /// 刷新令牌。
  String? refresh;

  /// 旧鉴权头。
  String? previousAccess;

  /// 旧刷新令牌。
  String? previousRefresh;

  /// 当前用户。
  String user = tkDioman;

  /// accessToken 的测试实现。
  /// 当前令牌。
  @override
  String? get accessToken => access;

  /// refreshToken 的测试实现。
  /// 当前刷新令牌。
  @override
  String? get refreshToken => refresh;

  /// replace 的测试实现。
  /// 更换令牌并记录变化。
  void replace(String? a, String? r) {
    if (a != access) previousAccess = access;
    if (r != refresh) previousRefresh = refresh;
    access = a;
    refresh = r;
  }

  /// accept 的测试实现。
  /// 接收登录或刷新信封数据。
  void accept(Map<String, dynamic> data) => replace(
      data[tkAccessToken] == null ? null : 'Bearer ${data[tkAccessToken]}',
      data[tkRefreshToken] as String?);

  /// 写入刷新后的原始凭证。
  @override
  void setTokens({required String accessToken, String? refreshToken}) =>
      replace(accessToken, refreshToken);

  /// clear 的测试实现。
  /// 清空当前令牌。
  @override
  void clear() => replace(null, null);

  /// mutate 的测试实现。
  /// 按契约篡改令牌。
  void mutate(Map<String, dynamic> spec) {
    final parts = (access ?? tkEmpty).split(tkSpace).last.split(tkDot);
    String? a = access;
    switch (spec[tkAccess]) {
      case tkNone:
        a = null;
      case tkEmpty6:
        a = tkEmpty;
      case tkGarbage:
        a = 'Bearer garbage-not-jwt';
      case tkPrevious:
        a = previousAccess;
      case 'wrong_scheme':
        a = 'Token ${(access ?? tkEmpty).split(tkSpace).last}';
      case 'bad_signature':
        if (parts.length == 3) {
          parts[2] =
              '${parts[2].startsWith(tkA) ? 'B' : tkA}${parts[2].substring(1)}';
        }
        a = 'Bearer ${parts.join(tkDot)}';
      case 'tampered_payload':
        if (parts.length == 3) {
          final payload = obj(jsonDecode(
              utf8.decode(base64Url.decode(base64Url.normalize(parts[1])))));
          payload[tkSub] = '${payload[tkSub]}-tampered';
          parts[1] = base64Url
              .encode(utf8.encode(jsonEncode(payload)))
              .replaceAll(tkEquals, tkEmpty);
        }
        a = 'Bearer ${parts.join(tkDot)}';
    }
    final r = switch (spec[tkRefresh]) {
      tkNone => null,
      tkGarbage => 'garbage-refresh',
      tkPrevious => previousRefresh,
      _ => refresh,
    };
    replace(a, r);
  }
}

/// 只读库返回的类别、原因和逻辑状态，不重复判定业务信封。
String classify(Object error) {
  if (error is! DiomanResponse) return 'unknown:$error';
  return switch (error.kind) {
    DiomanKind.ok => tkOk,
    DiomanKind.http => 'http:${error.status}',
    DiomanKind.biz => 'business:${error.status}',
    DiomanKind.network => tkNetwork,
    DiomanKind.timeout => tkTimeout,
    DiomanKind.cancel => tkCanceled,
    DiomanKind.parse => 'parse',
    DiomanKind.auth => error.reason ?? tkExpired,
    DiomanKind.breaker => 'breaker_open',
    DiomanKind.offline => tkOffline,
    DiomanKind.internal => 'internal',
  };
}

/// 封装 ScenarioRun 测试夹具。
/// 一轮场景的解释器、观测与生命周期。
class ScenarioRun {
  /// 构造独立测试实例。
  /// 每项独立创建代理与隔离标识。
  ScenarioRun(this.scenario, this.network, this.style, this.runId);

  /// 场景定义。
  final Map<String, dynamic> scenario;

  /// 起始网络。
  final String network;

  /// 起始响应风格。
  final String style;

  /// 唯一轮次标识。
  final String runId;

  /// 服务旁路客户端。
  final server = ioClient(serverUrl);

  /// netem 控制客户端。
  final netem = ioClient(netemUrl);

  /// 被测客户端。
  ScenarioClient? client;

  /// 仅保存本轮创建的代理标识。
  String? proxyId;

  /// 本轮代理地址。
  String? proxyBase;

  /// 所有在途取消令牌。
  final active = <CancelToken>{};

  /// 每次原生批量取消实际命中的令牌数。
  final cancelAllCounts = <int>[];

  /// 后台步骤。
  final fired = <Future<void>>[];

  /// 请求最终事实。
  final records = <String, dynamic>{};

  /// 请求完成顺序。
  final order = <String>[];

  /// 失败原因。
  final failures = <String>[];

  /// 服务端最终观测。
  Map<String, dynamic> observation = {};

  /// 代理最终统计。
  Map<String, dynamic> proxyStats = {};

  /// 超时后停止解释新步骤。
  bool stopped = false;

  /// CPU 熔断命中后停止整个验收。
  bool cpuBlocked = false;

  /// 控制面或服务观测不可达时停止领取新场景。
  bool infrastructureUnavailable = false;

  /// stop 的测试实现。
  /// 中断本轮逻辑请求，不操作任何服务进程。
  void stop() {
    stopped = true;
    for (final token in active.toList()) {
      token.cancel('testkit interrupted');
    }
  }

  /// merge 的测试实现。
  /// 合并预期，嵌套字段递归覆盖。
  static Map<String, dynamic> merge(
          Map<String, dynamic> a, Map<String, dynamic> b) =>
      {
        ...a,
        for (final e in b.entries)
          e.key: e.value is Map && a[e.key] is Map
              ? merge(obj(a[e.key]), obj(e.value))
              : e.value,
      };

  /// expectation 的测试实现。
  /// 当前网络和风格的预期。
  Map<String, dynamic> get expectation {
    final all = obj(scenario[tkExpect]);
    var out = <String, dynamic>{};
    for (final key in ['*', '*@$style', network, '$network@$style']) {
      out = merge(out, obj(all[key]));
    }
    return out;
  }

  /// observe 的测试实现。
  /// 旁路读取本轮观测。
  Future<Map<String, dynamic>> observe() async =>
      obj(obj((await server.get<dynamic>('/t/runs/$runId',
              options: Options(headers: {runHeader: runId})))
          .data)[tkData]);

  /// bypass 的测试实现。
  /// 旁路环境准备请求，不经过被测插件链。
  Future<Map<String, dynamic>> bypass(String path,
      {dynamic data, String method = tkPOST}) async {
    final r = await server.request<dynamic>(path,
        data: data, options: Options(method: method, headers: client!.headers));
    if ((r.statusCode ?? 0) >= 400) throw StateError('旁路 $path 失败: ${r.data}');
    return obj(obj(r.data)[tkData]);
  }

  /// setNetwork 的测试实现。
  /// 新建或切换自己拥有的代理。
  Future<void> setNetwork(dynamic value) async {
    if (value == tkDns) {
      client?.dio.options.baseUrl = tkHttpNonexistentInvalid47080;
      return;
    }
    final profile = value == tkFlap ? tkOffline : value;
    if (proxyId == null) {
      final p = obj((await netem.post<dynamic>(tkProxies,
              data: {tkUpstream: upstream, tkProfile: profile}))
          .data);
      proxyId = '${p[tkId]}';
      proxyBase = 'http://127.0.0.1:${p[tkPort]}';
      final journal = File(tkAcceptResultsProxiesNdjson);
      journal.parent.createSync(recursive: true);
      journal.writeAsStringSync(
          '${jsonEncode({
                tkRunId: runId,
                tkCreated: p,
                tkControl: netemUrl,
                tkUpstream: upstream
              })}\n',
          mode: FileMode.append);
    } else {
      await netem.put<dynamic>('/proxies/$proxyId', data: {tkProfile: profile});
    }
    if (value == tkFlap) {
      await netem.post<dynamic>('/proxies/$proxyId/schedule', data: [
        {'at_ms': 1500, tkProfile: tkNormal}
      ]);
    }
    client?.dio.options.baseUrl = proxyBase!;
  }

  /// probe 的测试实现。
  /// 对探测能力提供真实连通性输入。
  Future<void> probe(CancelToken token) async {
    final c = client!;
    if (!c.plugins.containsKey(tkOffline)) return;
    final d = ioClient(c.dio.options.baseUrl,
        budget: network == tkFlap ? flapProbeBudget : defaultBudget);
    try {
      await d.get<dynamic>(
          obj(c.plugins[tkOffline])[tkProbe] as String? ?? tkTPing,
          options: Options(headers: c.headers),
          cancelToken: token);
      c.online = true;
    } on DioException {
      c.online = false;
    } finally {
      d.close(force: true);
    }
    c.connectivity.add(c.online);
  }

  /// request 的测试实现。
  /// 发起一个逻辑请求，错误记录为事实，映射错误仍向上传播。
  Future<void> request(Map<String, dynamic> spec, String suffix) async {
    final c = client!;
    final id = '${spec[tkId]}$suffix';
    final options = obj(spec[tkOptions]);
    if (stopped || serviceBlocked) return;
    var token = CancelToken();
    active.add(token);
    final requestStyle = c.style;
    final rec = <String, dynamic>{tkId: id, tkStyle: requestStyle};
    final before = obj(obj(expectation[tkResults])[id]).containsKey(tkFromCache)
        ? await observe()
        : null;
    Timer? cancelTimer;
    if (spec[tkCancelAfterMs] != null) {
      cancelTimer = Timer(ms(spec[tkCancelAfterMs] as num),
          () => token.cancel('testkit cancel'));
    }
    try {
      await probe(token);
      final r = await c.api.send<dynamic>(
          spec[tkMethod] as String? ?? tkGET, spec[tkPath] as String,
          query: obj(spec[tkQuery]),
          data: requestBody(spec[tkBody]),
          call: c.call(options, token: token, headers: obj(spec['headers'])));
      rec.addAll({tkOk: r.ok, tkData: r.data, tkStatus: r.status});
      if (!r.ok) rec.addAll({tkError: classify(r), tkMessage: r.message});
    } catch (e) {
      rec.addAll({tkOk: false, tkError: classify(e), tkMessage: '$e'});
      if (e is DioException &&
          e.response?.statusCode == 503 &&
          '${e.response?.data}'.toLowerCase().contains('cpu')) {
        cpuBlocked = true;
      }
    } finally {
      cancelTimer?.cancel();
      active.remove(token);
      records[id] = rec;
      order.add(id);
    }
    if (before != null) {
      final after = await observe();
      rec[tkReached] =
          (after[tkTotal] as int? ?? 0) > (before[tkTotal] as int? ?? 0);
    }
  }

  /// step 的测试实现。
  /// 解释契约全部十六种 step，组合步骤递归执行。
  Future<void> step(Map<String, dynamic> s, [String suffix = tkEmpty]) async {
    if (stopped || serviceBlocked) return;
    if (s.length != 1) throw StateError('step 必须只有一个键');
    final k = s.keys.single;
    final v = s[k];
    final c = client!;
    switch (k) {
      case tkRequest:
        await request(obj(v), suffix);
      case 'sequence':
        for (final x in v as List) {
          await step(obj(x), suffix);
        }
      case tkParallel:
        await Future.wait((v as List).map((x) => step(obj(x), suffix)));
      case 'fire':
        fired.add(
            step(obj(v), suffix).catchError((Object error, StackTrace stack) {
          failures.add('fire 步骤失败: $error\n$stack');
        }));
      case 'repeat':
        final r = obj(v);
        for (var i = 0; i < (r['times'] as int); i++) {
          await step(
              obj(r['step']), r['id_suffix'] == false ? suffix : '$suffix#$i');
        }
      case tkWait:
        await pause(v as num);
      case tkLogin:
        final spec = obj(v);
        c.tokens.user = spec[tkUser] as String? ?? tkDioman;
        c.tokens.accept(await bypass(loginPath,
            data: {...spec, tkUsername: c.tokens.user}));
      case tkExpireLocal:
        final expired = await bypass(loginPath,
            data: {tkUsername: c.tokens.user, 'access_ttl_ms': 1});
        c.tokens.replace('Bearer ${expired[tkAccessToken]}', c.tokens.refresh);
        await pause(30);
      case 'set_token':
        c.tokens.mutate(obj(v));
      case 'logout':
        await bypass('/t/auth/logout',
            data: {tkRefreshToken: c.tokens.refresh});
        c.tokens.clear();
      case 'revoke':
        await bypass('/t/auth/revoke', data: {
          tkUsername: obj(v)[tkUser] ?? c.tokens.user,
          tkScope: obj(v)[tkScope] ?? 'all'
        });
      case tkCancelAll:
        await pause(obj(v)['after_ms'] as num? ?? 0);
        cancelAllCounts.add(c.api.cancelAll());
      case 'clear_cache':
        await c.api.plugin<DiomanCacheApi>()?.clear();
      case tkNetwork:
        await setNetwork(v);
      case tkStyle:
        c.style = v as String;
      case 'server_reset':
        await bypass('/t/runs/$runId', method: tkDELETE);
      default:
        throw StateError('未知 step: $k');
    }
  }

  /// checkLocal 的测试实现。
  /// 校验全部请求、客户端事件、顺序与持续时间。
  void checkLocal(int duration) {
    final exp = expectation;
    final c = client!;
    for (final e in obj(exp[tkResults]).entries) {
      final actual = obj(records[e.key]);
      final wanted = obj(e.value);
      if (actual.isEmpty) {
        failures.add('${e.key}: 缺少请求结果');
        continue;
      }
      if (wanted.containsKey(tkOk) && wanted[tkOk] != actual[tkOk]) {
        failures
            .add('${e.key}: 预期 ok=${wanted[tkOk]}，实际 ${jsonEncode(actual)}');
      }
      if (wanted.containsKey(tkError)) {
        final errors = wanted[tkError] is List
            ? wanted[tkError] as List
            : [wanted[tkError]];
        final match = errors.any((x) {
          final expected = '$x'.startsWith('status:')
              ? '${actual[tkStyle] == tkHttp ? tkHttp : tkBusiness}:${'$x'.substring(7)}'
              : x;
          return actual[tkOk] == false &&
              (expected == 'any' || expected == actual[tkError]);
        });
        if (!match) {
          failures
              .add('${e.key}: 预期错误 $errors，实际 ${actual[tkError] ?? tkEmpty5}');
        }
      }
      if (wanted.containsKey(tkData) &&
          !subset(actual[tkData], wanted[tkData])) {
        failures.add('${e.key}: data 子集不符');
      }
      if (wanted.containsKey(tkHit) &&
          obj(obj(actual[tkData])[tkMeta])[tkHit] != wanted[tkHit]) {
        failures.add(
            '${e.key}: hit 不符，实际 ${obj(obj(actual[tkData])[tkMeta])[tkHit]}，预期 ${wanted[tkHit]}');
      }
      if (wanted.containsKey(tkSameAs)) {
        final reason = wanted[tkOk] != true
            ? '${e.key}: same_as 只能用于 ok:true 的结果'
            : checkSameAs(e.key, records, obj(wanted[tkSameAs]));
        if (reason != null) failures.add(reason);
      }
      if (wanted.containsKey(tkFromCache) &&
          actual[tkReached] != !(wanted[tkFromCache] as bool)) {
        failures.add('${e.key}: from_cache 无服务端证据或不符');
      }
    }
    if (exp[tkOrder] != null && !subset(order, exp[tkOrder])) {
      failures.add('order=${jsonEncode(order)}，预期 ${jsonEncode(exp[tkOrder])}');
    }
    range(failures, tkDurationMs, duration, exp[tkDurationMs]);
    final events = eventSnapshot();
    for (final e in obj(exp[tkClient]).entries) {
      if (e.key == tkLoading) {
        if (!subset(c.loadingEvents, e.value)) {
          failures.add('loading=${c.loadingEvents}，预期 ${e.value}');
        }
      } else {
        range(failures, 'client.${e.key}', events[e.key] as num, e.value);
      }
    }
    if (c.api.pending != 0) failures.add('pending_after=${c.api.pending}');
    var balance = 0;
    for (final value in c.loadingEvents) {

def spread(values):
    """计算三轮相对最小值的最大离散度（百分比）。"""
