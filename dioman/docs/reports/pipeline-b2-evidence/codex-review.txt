import 'package:dio/dio.dart';

/// 递归复制配置容器；不透明对象保留身份。
dynamic cloneDiomanConfig(dynamic value) {
  if (value is Map) {
    return value.map((k, v) => MapEntry(k, cloneDiomanConfig(v)));
  }
  if (value is List) return value.map(cloneDiomanConfig).toList();
  return value;
}

/// 克隆配置及其三个可变容器，返回与源隔离的配置。
BaseOptions cloneDiomanOptions(BaseOptions source) => source.copyWith(
    headers: Map<String, dynamic>.from(cloneDiomanConfig(source.headers)),
    queryParameters:
        Map<String, dynamic>.from(cloneDiomanConfig(source.queryParameters)),
    extra: Map<String, dynamic>.from(cloneDiomanConfig(source.extra)));

/// 写入非空、非默认的配置；容器按顶层键合并并深拷贝。
/// BaseOptions 无法区分显式默认值与省略，因此空地址和 json 保留宿主值。
void mergeDiomanOptions(BaseOptions target, BaseOptions source) {
  final c = cloneDiomanOptions(source);
  final host = cloneDiomanOptions(target);
  if (c.baseUrl.isNotEmpty) target.baseUrl = c.baseUrl;
  if (c.connectTimeout != null) target.connectTimeout = c.connectTimeout;
  if (c.receiveTimeout != null) target.receiveTimeout = c.receiveTimeout;
  if (c.sendTimeout != null) target.sendTimeout = c.sendTimeout;
  target.headers = {...host.headers, ...c.headers};
  target.queryParameters = {...host.queryParameters, ...c.queryParameters};
  target.extra = {...host.extra, ...c.extra};
  if (c.responseType != ResponseType.json) target.responseType = c.responseType;
}

/// 装上即过滤 query 与 Map 请求体的顶层空字段。
class DiomanFilter<R extends DiomanResponse<dynamic>> extends DiomanPlugin<R> {
  /// 创建过滤器，例如 `DiomanFilter(ignoreKeys: ['nullable'])`；特殊载荷原样放行。
  const DiomanFilter(
      {this.predicate = defaultPredicate,
      this.ignoreKeys = const [],
      this.ignoreValues = const [],
      this.enabled = true});

  /// 字段删除规则。
  final DiomanFilterPredicate predicate;

  /// 保留键。
  final List<String> ignoreKeys;

  /// 保留值。
  final List<Object?> ignoreValues;

  /// 安装期开关。
  @override
  final bool enabled;

  /// 稳定名称。
  static const pluginName = 'dioman:filter';

  /// 请求级 bool 或 DiomanFilterOptions 键。
  static final option = DiomanOptionKey<DiomanFilterOptions>(pluginName);

  /// 默认删除 null、NaN、空白字符串，保留零和 false。
  static bool defaultPredicate(String key, dynamic value) =>
      value == null ||
      (value is num && value.isNaN) ||
      (value is String && value.trim().isEmpty);

  /// 返回稳定名称。
  @override
  String get name => pluginName;

  /// 请求整形阶段。
  @override
  DiomanStage get stage => DiomanStage.prepare;

  /// 复制豁免集合，避免外部修改基础配置。
  @override
  DiomanPluginInstance<R> setup(Dioman<R> client) =>
      _Filter<R>(DiomanFilterOptions(
          predicate: predicate,
          ignoreKeys: List.of(ignoreKeys),
          ignoreValues: List.of(ignoreValues)));
}

/// 每客户端的过滤配置。
class _Filter<R extends DiomanResponse<dynamic>>
    extends DiomanPluginInstance<R> {
  /// 保存基础配置。
  _Filter(this.base);

  /// 预合并配置。
  final DiomanFilterOptions base;

  /// 判定一次并仅在删除字段时派生请求。
  @override
  Future<DiomanResponse<dynamic>> handle(
      DiomanContext<R> ctx, DiomanNext<R> next) {
    final o = resolveOption<DiomanFilterOptions>(
        ctx.request.call.options[DiomanFilter.option],
        base: base,
        merge: (a, b) => DiomanFilterOptions(
            enabled: b.enabled,
            predicate: b.predicate ?? a.predicate,
            ignoreKeys: [...a.ignoreKeys!, ...?b.ignoreKeys],
            ignoreValues: [...a.ignoreValues!, ...?b.ignoreValues]));
    if (o == null || o.enabled == false) return next();

    /// 豁免项优先于用户谓词。
    bool drop(dynamic k, dynamic v) =>
        !o.ignoreKeys!.contains('$k') &&
        !o.ignoreValues!.contains(v) &&
        o.predicate!('$k', v);
    final queryKeys = ctx.request.query.entries
        .where((e) => drop(e.key, e.value))
        .map((e) => e.key)
        .toSet();
    final data = ctx.request.data;
    if (data is Map) {
      final body = Map.of(data)..removeWhere(drop);
      if (body.length != data.length) {
        return next(RequestPatch(removeQuery: queryKeys, data: body));
      }
    }
    if (queryKeys.isEmpty) return next();
    return next(RequestPatch(removeQuery: queryKeys));
  }
}

      states.values.fold(0, (sum, state) => sum + state.count);

  /// 取消上一个边沿窗口。
  void clear(_Display state) {
    state.timer?.cancel();
    state.timer = null;
  }

  /// 先更新状态，再通知；回调异常不能破坏配对。
  void schedule(void Function(bool) fn, _Display state, bool visible,
      Duration delay, DiomanLogger log) {
    /// 发出当前显示边沿。
    void fire() {
      state.timer = null;
      state.shown = visible;
      if (!visible) states.remove(fn);
      try {
        fn(visible);
      } catch (e, s) {
        log('[${DiomanLoading.pluginName}] 通知回调失败: $e\n$s');
      }
    }

    if (delay > Duration.zero) {
      state.timer = clock.createTimer(delay, fire);
    } else {
      fire();
    }
  }

  /// 逻辑请求计数严格配对，在 finally 中释放。
  @override
  Future<DiomanResponse<dynamic>> handle(
      DiomanContext<R> ctx, DiomanNext<R> next) async {
    final o = resolveOption<DiomanLoadingOptions>(
        ctx.request.call.options[DiomanLoading.option],
        base: base,
        merge: (a, b) => DiomanLoadingOptions(
            enabled: b.enabled,
            onChanged: b.onChanged ?? a.onChanged,
            delay: b.delay ?? a.delay,
            delayClose: b.delayClose ?? a.delayClose));
    if (disposed || o == null || o.enabled == false) return next();
    final fn = o.onChanged!;
    final state = states.putIfAbsent(fn, () => _Display(ctx.log));
    if (++state.count == 1) {
      clear(state);
      if (!state.shown) schedule(fn, state, true, o.delay!, ctx.log);
    }
    try {
      final result = await next();
      final cancelled = ctx.cancelled();
      return result.kind == DiomanKind.internal ? result : cancelled ?? result;
    } finally {
      if (--state.count == 0 && !disposed) {
        clear(state);
        if (state.shown) {
          schedule(fn, state, false, o.delayClose!, ctx.log);
        } else {
          states.remove(fn);
        }
      }
    }
  }

  /// 取消窗口并立即关闭已显示回调，在途请求不再通知。
  @override
  void dispose() {
    disposed = true;
    for (final entry in states.entries.toList()) {
      clear(entry.value);
      if (entry.value.shown) {
        schedule(entry.key, entry.value, false, Duration.zero, entry.value.log);
      }
    }
    states.clear();
  }
}

class DiomanNotify<R extends DiomanResponse<dynamic>> extends DiomanPlugin<R> {

  /// 例如 `DiomanNotify(notify: (message, r) => print(message))`。
  const DiomanNotify(
      {required this.notify,
      this.stringify = defaultStringify,
      this.enabled = true});

  /// 通知回调，异常隔离并记日志。
  final void Function(String message, DiomanResponse<dynamic> result) notify;

  /// 消息生成回调，与 notify 共享通知异常边界。
  final String Function(DiomanResponse<dynamic> result, DiomanContext<R> ctx)
      stringify;

  /// 安装期开关。
  @override
  final bool enabled;

  /// 稳定名称。
  static const pluginName = 'dioman:notify';

  /// 请求级只接受 bool；false 关闭。
  static final option = DiomanOptionKey<bool>(pluginName);

  /// 默认忽略成功、取消、静默请求，否则取消息或 kind/status。
  static String defaultStringify(
          DiomanResponse<dynamic> r, DiomanContext ctx) =>
      r.ok || r.kind == DiomanKind.cancel || ctx.request.silent
          ? ''
          : (r.message.isEmpty ? '${r.kind.name} ${r.status}' : r.message);

  /// 返回稳定名称。
  @override
  String get name => pluginName;

  /// 最外层结果观察阶段。
  @override
  DiomanStage get stage => DiomanStage.outer;

  /// 建立客户端实例。
  @override
  DiomanPluginInstance<R> setup(Dioman<R> client) => _Notify<R>(this);
}

/// 不持有跨请求状态的通知实例。
class _Notify<R extends DiomanResponse<dynamic>>
    extends DiomanPluginInstance<R> {
  /// 保存配置。
  _Notify(this.config);

  /// 插件配置。
  final DiomanNotify<R> config;

  /// 等待最终结果后重查取消，通知异常不改变结果。
  @override
  Future<DiomanResponse<dynamic>> handle(
      DiomanContext<R> ctx, DiomanNext<R> next) async {
    final on = resolveOption<bool>(
        ctx.request.call.options[DiomanNotify.option],
        base: true,
        merge: identical);
    final r = await next();
    final cancelled = ctx.cancelled();
    if (on == null) return r;
    // internal 优先，取消不掩盖程序错误。
    final result = r.kind == DiomanKind.internal ? r : cancelled ?? r;
    try {
      final text = config.stringify(result, ctx);
      if (text.isNotEmpty) config.notify(text, result);
    } catch (e, s) {
      ctx.log('[${DiomanNotify.pluginName}] 通知回调失败: $e\n$s');
    }
    return r;
  }
}

/// 路径替换的请求级覆盖；null 字段继承插件配置。
class DiomanRepathOptions {
  /// [pattern] 的第一个非空捕获组是键名；[removeKey] 控制消费字段删除。
  const DiomanRepathOptions({this.enabled, this.pattern, this.removeKey});

  /// false 跳过本次处理。
  final bool? enabled;

  /// 占位符正则。
  final RegExp? pattern;

  /// 是否移除被消费的字段。
  final bool? removeKey;

/// 只替换 URL 路径中的变量，query 优先于 Map 请求体。
class DiomanRepath<R extends DiomanResponse<dynamic>> extends DiomanPlugin<R> {
  /// 创建插件，例如 `DiomanRepath(removeKey: false)`；值按路径段编码。
  const DiomanRepath(
      {this.pattern, this.removeKey = true, this.enabled = true});

  /// 自定义捕获正则，省略时匹配三种内置语法。
  final RegExp? pattern;

  /// 替换后删除来源键。
  final bool removeKey;

  /// 安装期开关。
  @override
  final bool enabled;

  /// 稳定名称。
  static const pluginName = 'dioman:repath';

  /// 请求选项键，接受 bool 或 DiomanRepathOptions。
  static final option = DiomanOptionKey<DiomanRepathOptions>(pluginName);

  /// 默认占位符模式。
  static final defaultPattern = RegExp(r'\{([^}]+)\}|\[([^\]]+)]|:([^/?#\s]+)');

  /// 返回稳定名称。
  @override
  String get name => pluginName;

  /// 请求整形阶段。
  @override
  DiomanStage get stage => DiomanStage.prepare;

  /// 预合并默认配置。
  @override
  DiomanPluginInstance<R> setup(Dioman<R> client) =>
      _Repath<R>(DiomanRepathOptions(
          pattern: pattern ?? defaultPattern, removeKey: removeKey));
}

/// 每客户端的已解析配置。
class _Repath<R extends DiomanResponse<dynamic>>
    extends DiomanPluginInstance<R> {
  /// 保存基础配置。
  _Repath(this.base);

  /// 预合并配置。
  final DiomanRepathOptions base;

  /// 生成路径与来源字段的补丁，不修改原请求。
  @override
  Future<DiomanResponse<dynamic>> handle(
      DiomanContext<R> ctx, DiomanNext<R> next) {
    final o = resolveOption<DiomanRepathOptions>(
        ctx.request.call.options[DiomanRepath.option],
        base: base,
        merge: (a, b) => DiomanRepathOptions(
            enabled: b.enabled,
            pattern: b.pattern ?? a.pattern,
            removeKey: b.removeKey ?? a.removeKey));
    if (o == null || o.enabled == false) return next();
    final original = ctx.request.path;
    final origin =
        RegExp(r'^[a-zA-Z][a-zA-Z0-9+.-]*://[^/]*').firstMatch(original)?.end ??
            0;
    final rest = original.substring(origin);
    final suffixAt = rest.indexOf(RegExp(r'[?#]'));
    final path = suffixAt < 0 ? rest : rest.substring(0, suffixAt);
    final suffix = rest.substring(path.length);
    final query = ctx.request.query;
    final data = ctx.request.data;
    final usedQuery = <String>{};
    final usedData = <String>{};
    final replaced = path.replaceAllMapped(o.pattern!, (m) {
      String? key;
      for (var i = 1; i <= m.groupCount; i++) {
        final group = m.group(i);
        if (group != null && group.isNotEmpty) {
          key = group;
          break;
        }
      }
      if (key == null) return m[0]!;
      if (query.containsKey(key)) {
        usedQuery.add(key);
        return Uri.encodeComponent('${query[key]}');
      }
      if (data is Map && data.containsKey(key)) {
        usedData.add(key);
        return Uri.encodeComponent('${data[key]}');
      }
      return m[0]!;
    });
    if (replaced == path) return next();
    final newPath = original.substring(0, origin) + replaced + suffix;
    if (!o.removeKey!) return next(RequestPatch(path: newPath));
    if (usedData.isEmpty) {
      return next(RequestPatch(path: newPath, removeQuery: usedQuery));
    }
    final body = Map.of(data as Map)
      ..removeWhere((k, _) => usedData.contains(k));
    return next(
        RequestPatch(path: newPath, removeQuery: usedQuery, data: body));
  }
}

/// 插件校验错误的参数名。
const _nameArgument = 'name';

/// 编译后的上下文执行器。
typedef DiomanRunner<R extends DiomanResponse<dynamic>>
    = Future<DiomanResponse<dynamic>> Function(DiomanContext<R> ctx);

/// 已安装的配置及实例。
class DiomanRegistration<R extends DiomanResponse<dynamic>> {
  /// 保存配置与客户端实例。
  const DiomanRegistration(this.plugin, this.instance);

  /// 插件配置。
  final DiomanPlugin<R> plugin;

  /// setup 返回的独立状态。
  final DiomanPluginInstance<R>? instance;
}

/// 按名称确定位置；自定义插件位于同阶段内置插件前后。
(int, int) _position(DiomanPlugin plugin) {
  for (final entry in diomanBuiltinStages.entries) {
    final index = entry.value.indexOf(plugin.name);
    if (index >= 0) return (entry.key.index, index);
  }
  if (plugin.name.startsWith('dioman:')) {
    throw ArgumentError.value(plugin.name, _nameArgument, '保留前缀');
  }
  return (plugin.stage.index, plugin.enforce == DiomanEnforce.pre ? -1 : 100);
}

/// 在 setup 前验证重名和保留前缀，拒绝非法配置且不产生副作用。
void validateDiomanPlugins(Iterable<DiomanPlugin> plugins) {
  final names = <String>{};
  for (final plugin in plugins) {
    if (!names.add(plugin.name)) {
      throw ArgumentError.value(plugin.name, _nameArgument, '插件重名');
    }
    if (!diomanSetupPlugins.contains(plugin.name)) _position(plugin);
  }
}

/// 稳定排序已启用且有 handle 的插件；不修改输入。
List<DiomanRegistration<R>>
    sortDiomanPipeline<R extends DiomanResponse<dynamic>>(
        List<DiomanRegistration<R>> registrations) {
  final indexed =
      registrations.indexed.where((e) => e.$2.instance != null).toList();
  indexed.sort((a, b) {
    final x = _position(a.$2.plugin);
    final y = _position(b.$2.plugin);
    final stage = x.$1.compareTo(y.$1);
    if (stage != 0) return stage;
    final order = x.$2.compareTo(y.$2);
    return order == 0 ? a.$1.compareTo(b.$1) : order;
  });
  return indexed.map((e) => e.$2).toList();
}

/// 从内到外编译一次；每层含一个异常边界，无补丁复用上下文。
DiomanRunner<R> compileDiomanPipeline<R extends DiomanResponse<dynamic>>(
    List<DiomanRegistration<R>> layers, DiomanRunner<R> transport) {
  var run = transport;
  for (final layer in layers.reversed) {
    final inner = run;
    run = (ctx) async {
      try {
        return await layer.instance!.handle(ctx, ([patch]) async {
          DiomanContext<R> target;
          try {
            target = patch == null ? ctx : ctx.derive(patch);
          } catch (e, s) {
            return ctx.fail(DiomanFailure(
                kind: DiomanKind.internal, error: e, errorStack: s));
          }
          return inner(target);
        });
      } catch (e, s) {
        return ctx.fail(
            DiomanFailure(kind: DiomanKind.internal, error: e, errorStack: s));
      }
    };
  }
  return run;
}

/// 关闭 debug 时复用的空日志器。
void _noLog(String _) {}

/// 以 Dio 为引擎的洋葱管线客户端；旧入口与本库完全独立。
class Dioman<R extends DiomanResponse<dynamic>> {
  /// 创建客户端，例如 `Dioman(dio: Dio(), debug: true)`。
  /// [response] 控制归一化，[reject] 只转换动词方法的可预期失败。
  Dioman(
      {required this.dio,
      List<DiomanPlugin<R>> plugins = const [],
      DiomanResponseFactory<R>? response,
      this.key = const DiomanKeyOptions(),
      this.reject,
      Object? debug,
      this.clock = const DiomanClock.system()})
      : response = response ?? DiomanResponseFactory<R>(),
        _debug = debug != null && debug != false,
        _logger =
            debug is DiomanLogger ? debug : (debug == true ? print : _noLog) {
    if (debug != null && debug is! bool && debug is! DiomanLogger) {
      throw ArgumentError.value(debug, 'debug');
    }
    validateDiomanPlugins(plugins);
    try {
      for (final plugin in plugins) {
        _registrations.add(DiomanRegistration(
            plugin, plugin.enabled ? plugin.setup(this) : null));
      }
      _compile();
    } catch (_) {
      for (final registration in _registrations.reversed) {
        try {
          registration.instance?.dispose();
        } catch (_) {/* 保留安装时首次异常。 */}
      }
      rethrow;
    }
  }

  /// 底层 Dio；用户拦截器在每次传输时运行。
  final Dio dio;

  /// 客户端的归一化服务。
  final DiomanResponseFactory<R> response;

  /// 客户端构键配置。
  final DiomanKeyOptions key;

  /// 可预期失败的最终异常转换器。
  final Object Function(R)? reject;

  /// 可注入时间服务。
  final DiomanClock clock;

  /// 是否记录调用点。
  final bool _debug;

  /// 用户日志器或默认输出。
  final DiomanLogger _logger;

  /// 注册顺序，供生命周期和 extend 使用。
  final _registrations = <DiomanRegistration<R>>[];

  /// 已排序的执行层快照。
  List<DiomanRegistration<R>> _layers = [];

  /// 编译后的链，在途请求捕获当前值。
  late DiomanRunner<R> _run;

  /// 活跃逻辑请求。
  final _pending = <DiomanContextState>{};

  /// 请求序号。
  int _sequence = 0;

  /// 生命周期状态。
  bool _disposed = false;

  /// 当前活跃逻辑请求数。
  int get pending => _pending.length;

  /// 活跃的内核取消等待订阅总数，供不变量检查。
  int get cancellationSubscriptions =>
      _pending.fold(0, (n, s) => n + s.listeners.length);

  /// 隔离日志回调异常，避免 debug 改变请求结果。
  void _log(String message) {
    try {
      _logger(message);
    } catch (_) {/* 日志通知不影响请求。 */}
  }

  /// 拒绝销毁后的操作。
  void _checkAlive() {
    if (_disposed) throw StateError('客户端已销毁');
  }

  /// 在配置变动时编译链，不在请求期间排序。
  void _compile() {
    _layers = sortDiomanPipeline(_registrations);
    _run = compileDiomanPipeline(_layers, (ctx) => diomanTransport(dio, ctx));
  }

  /// 添加插件；先校验再 setup，安装失败不修改当前链。
  void use(DiomanPlugin<R> plugin) {
    _checkAlive();
    validateDiomanPlugins([..._registrations.map((r) => r.plugin), plugin]);
    final instance = plugin.enabled ? plugin.setup(this) : null;
    _registrations.add(DiomanRegistration(plugin, instance));
    _compile();
  }

  /// 按 [name] 移除并释放插件；不存在返回 false，在途保留原链。
  bool eject(String name) {
    final index = _registrations.indexWhere((r) => r.plugin.name == name);
    if (index < 0) return false;
    final removed = _registrations.removeAt(index);
    _compile();
    removed.instance?.dispose();
    return true;
  }

  /// 返回实现 [A] 的第一个实例，例如 `api.plugin<CacheApi>()`。
  A? plugin<A>() {
    for (final r in _registrations) {
      if (r.instance is A) return r.instance as A;
    }
    return null;
  }

  /// 返回解析后的固定阶段和名称顺序，返回列表不可修改。
  List<(DiomanStage, String)> inspect() => List.unmodifiable(_layers.map((r) {
        final fixed = diomanBuiltinStages.entries
            .where((e) => e.value.contains(r.plugin.name));
        return (
          fixed.isEmpty ? r.plugin.stage : fixed.single.key,
          r.plugin.name
        );
      }));

  /// 中止尚未取消的逻辑请求，返回本次中止数。
  int cancelAll([Object? reason]) {
    var count = 0;
    for (final state in _pending.toList()) {
      if (!state.token.isCancelled) {
        count++;
        state.cancel(reason);
      }
    }
    return count;
  }

  /// 克隆 BaseOptions 并重新 setup 插件，状态不共享；[plugins] 替换继承列表。
  Dioman<R> extend({BaseOptions? options, List<DiomanPlugin<R>>? plugins}) {
    _checkAlive();
    final child = Dioman<R>(
        dio: Dio(cloneDiomanOptions(dio.options)),
        plugins: plugins ?? _registrations.map((r) => r.plugin).toList(),
        response: response,
        key: key,
        reject: reject,
        debug: _debug ? _logger : false,
        clock: clock);
    if (options != null) mergeDiomanOptions(child.dio.options, options);
    return child;
  }

  /// 先取消请求，再逆序释放全部实例；清理完毕后原栈抛出首个异常。
  void dispose() {
    if (_disposed) return;
    _disposed = true;
    cancelAll();
    final errors = <(Object, StackTrace)>[];
    for (final r in _registrations.reversed) {
      try {
        r.instance?.dispose();
      } catch (e, s) {
        errors.add((e, s));
      }
    }
    if (errors.isNotEmpty) {
      Error.throwWithStackTrace(errors.first.$1, errors.first.$2);
    }
  }

  /// 执行请求并返回结果；可预期失败不抛，internal 原异常原栈抛出。
  /// 例如 `await api.send<Object?>('GET', '/pets')`。
  Future<DiomanResponse<T>> send<T>(String method, String path,
      {Map<String, dynamic>? query, Object? data, DiomanCall? call}) async {
    final result =
        await _dispatch(method, path, query: query, data: data, call: call);
    if (result is DiomanResponse<T>) return result;
    return result.copy<T>();
  }

  /// 执行任意 [method] 并返回数据；可预期失败经 reject 后抛出。
  Future<T> request<T>(String method, String path,
      {Map<String, dynamic>? query, Object? data, DiomanCall? call}) async {
    final result =
        await _dispatch(method, path, query: query, data: data, call: call);
    if (!result.ok) throw (reject?.call(result as R) ?? result);
    return result.data as T;
  }

  /// 发送 GET 并返回数据；参数与 request 相同，例如 `api.get<Object?>('/pets')`。
  Future<T> get<T>(String path,
          {Map<String, dynamic>? query, Object? data, DiomanCall? call}) =>
      request<T>('GET', path, query: query, data: data, call: call);

  /// 发送 POST 并返回数据；参数与 request 相同，例如 `api.post<Object?>('/pets')`。
  Future<T> post<T>(String path,
          {Map<String, dynamic>? query, Object? data, DiomanCall? call}) =>
      request<T>('POST', path, query: query, data: data, call: call);

  /// 发送 PUT 并返回数据；参数与 request 相同，例如 `api.put<Object?>('/pets')`。
  Future<T> put<T>(String path,
          {Map<String, dynamic>? query, Object? data, DiomanCall? call}) =>
      request<T>('PUT', path, query: query, data: data, call: call);

  /// 发送 PATCH 并返回数据；参数与 request 相同，例如 `api.patch<Object?>('/pets')`。
  Future<T> patch<T>(String path,
          {Map<String, dynamic>? query, Object? data, DiomanCall? call}) =>
      request<T>('PATCH', path, query: query, data: data, call: call);

  /// 发送 DELETE 并返回数据；参数与 request 相同，例如 `api.delete<Object?>('/pets')`。
  Future<T> delete<T>(String path,
          {Map<String, dynamic>? query, Object? data, DiomanCall? call}) =>
      request<T>('DELETE', path, query: query, data: data, call: call);

  /// 发送 HEAD 并返回数据；参数与 request 相同，例如 `api.head<Object?>('/pets')`。
  Future<T> head<T>(String path,
          {Map<String, dynamic>? query, Object? data, DiomanCall? call}) =>
      request<T>('HEAD', path, query: query, data: data, call: call);

  /// 入口仅 compose 一次；登记后所有可能抛错的构造都在 finally 保护内。
  Future<DiomanResponse<dynamic>> _dispatch(String method, String path,
      {Map<String, dynamic>? query, Object? data, DiomanCall? call}) async {
    _checkAlive();
    final run = _run;
    final site = _debug ? StackTrace.current : null;
    final state = DiomanContextState(CancelToken());
    DiomanCancelLink? userLink;
    _pending.add(state);
    try {
      final c = call ?? const DiomanCall();
      if (c.cancelToken != null) {
        userLink = DiomanCancelLink(c.cancelToken!, state.cancel);
      }
      final options = Options(
              method: method,
              headers: c.headers,
              sendTimeout: c.sendTimeout,
              receiveTimeout: c.receiveTimeout)
          .compose(dio.options, path, queryParameters: query, data: data);
      if (c.connectTimeout != null) options.connectTimeout = c.connectTimeout;
      final ctx = DiomanContext<R>(
          id: ++_sequence,
          client: this,
          request: DiomanRequest(options, call: c),
          state: state,
          factory: response,
          keyService: DiomanKeyService(key),
          clock: clock,
          log: _log);
      var result = await run(ctx);
      if (result.kind != DiomanKind.internal) {
        result = ctx.cancelled() ?? result;
      }
      if (!result.ok && site != null && result.kind != DiomanKind.internal) {
        result = response.withCallSite(result, site);
      }
      if (result.kind == DiomanKind.internal) {
        Error.throwWithStackTrace(
            result.error ?? StateError('internal result without error'),
            result.errorStack ?? StackTrace.current);
      }
      return result;
    } finally {
      userLink?.dispose();
      state.dispose();
      _pending.remove(state);
    }
  }

/// 插件请求级选项键；按对象身份隔离，建议使用静态 final 实例。
class DiomanOptionKey<T> {
  /// [description] 仅供调试，例如 `DiomanOptionKey<int>('retry')`。
  DiomanOptionKey(this.description);

  /// 调试说明。
  final String description;
}

/// 解析统一选项；省略/true 复用 [base]，false 关闭；[optIn] 控制省略时关闭。
/// [merge] 合并类型化覆盖，[shorthand] 返回简写解析值；未知类型抛 ArgumentError。
/// 示例：`resolveOption<int>(true, base: 3, merge: (a, b) => b)`。
T? resolveOption<T>(Object? value,
    {required T base,
    required T Function(T base, T override) merge,
    T? Function(Object value)? shorthand,
    bool optIn = false}) {
  if (value == false || (value == null && optIn)) return null;
  if (value == null || value == true) return base;
  if (value is T) return merge(base, value as T);
  final short = shorthand?.call(value);
  if (short != null) return merge(base, short);
  throw ArgumentError.value(value, 'value', '不支持的插件选项');
}

s`，share 的策略枚举）。不认识的类型按 bug 抛 `ArgumentError`。插件级配置在 `setup` 时预合并；请求级 `true` 或省略时复
用同一份。
  4. **装上即生效**，只有改变数据新鲜度的 **cache、share 要在请求级开启**（optIn）。
  5. **改请求只用 `next(patch)`**，不改 `ctx.request`。
  6. 用户回调抛错：通知类隔离并输出错误日志；判定类按 bug 处理；`onRefresh` 抛错算刷新失败（见核心架构 §7）。
  7. **每个 await 之后用 `ctx.cancelled()` 重查取消**；等待一律用 `ctx.wait`。
  8. 会重发请求的插件（retry、auth、mock、share 的 retry 策略）在第一次 `next()` 之前调用 `ctx.markReplayable()`。
  
  ## 1. outer
  
  ### loading
  - 选项：`onChanged(bool visible)` 必填；`delay`（0→1 之后延迟显示，期间回落就不显示）、`delayClose`（1→0 之后延迟隐藏
，期间有新请求就不隐藏），默认都是 `Duration.zero`。请求级：`false`、`true`、`DiomanLoadingOptions`。
  - 模型：`final r = await next(); 通知; return r;`。每个逻辑请求最多一次，天然只在最终失败时通知，刷新成功不通知。
  - 内核依赖：outer 阶段、`request.silent`。
  
  ## 2. prepare
  
> ### repath
> - 选项不变：`pattern`（默认匹配 `{id}`、`:id`、`[id]`）、`removeKey`（默认 true）。只处理路径部分，先找 query（含 Bas
eOptions 的 queryParameters）再找 Map 形式的 data；值按路径段编码；同名占位符多次出现全部替换。算出新的 path/query/data
 后 `next(patch)`；没有变化直接 `next()`。
  
  ### filter
  - 选项：`predicate(key, value)`，默认剔除 null、NaN、空白字符串，保留 0 与 false；`ignoreKeys`、`ignoreValues`。剥 qu
ery 和 Map 形式的 data 的顶层字段；FormData、Stream 原样。没有可剔除的字段时不打补丁。
  - 变化：**装上即生效**（0.7 也是），默认谓词新增剔除 NaN（testkit 中立语义，且 `jsonEncode(double.nan)` 会抛错）。
  
  ### timeout
  - 选项不变：`timeouts`（按 `NetworkQuality` 档位的 connect/receive/send）、`probe`（返回当前网络档位）。请求级：`fals
e`、`DiomanTimeoutOptions`。
  - 模型：读出本次档位对应的超时，`next(patch)` 写入请求的超时字段。只改配置，真正计时由 Dio 执行（假设 H5，已在本地回
环核对）。
  
  ## 3. attempt
  
  ### retry
  - 选项沿用 0.7：`max`、`methods`（硬否决）、`statusCodes`（默认 408/429/500/502/503/504，读**逻辑状态** `r.status`）
、`shouldRetry(R r, int attempt)`、`delay`、`jitter`、`delayMax`、`respectRetryAfter`（默认 true，命中 `afterStatusCode
s` 默认 413/429/503 时按 `Retry-After` 的秒数或 HTTP 日期等待，上限 `retryAfterMax`，不叠 jitter、不受 delayMax 限制）
、`onRetry(attempt)`。请求级：整数 = max、`false`、`true`、`DiomanRetryOptions`。
  - 判定顺序：请求级关闭 → 方法不在白名单 → `kind` 属于 cancel、auth、internal、parse、breaker、offline 时不重试 → `sho
uldRetry` 返回布尔就采用 → network、timeout 重试 → `statusCodes` 包含 `r.status` 时重试。
  - 模型：`markReplayable()`；`var r = await next(); for (attempt = 1..max) { 判定; 等待 ctx.wait(delay); 取消则返回; r
 = await next(); } return r;`。次数用完返回最后一次结果，由落定决定抛不抛。每次尝试都会穿过 breaker、auth 和 source 阶
段。
  - 删除：重放交接、认领判断（`_notAdmitted`）、`share`/`cancel` 兼容 setter、`breaker` 字段。
  
  ### breaker
  ### offline
  - 选项沿用 0.7：`isOnline()`、`onConnectivityChanged`、`shouldQueue(request)`、`maxQueueSize`、`maxWait`。请求级：`fa
lse`、`DiomanOfflineOptions`。
  - 模型：在线 → `next()`。离线：不排队或队列已满 → `ctx.fail(kind: offline, reason: …)`；排队 → `ctx.wait(恢复在线, ti
meout: maxWait)`，恢复后 `next()`，超时返回 `kind: offline, reason: 'timeout'`，取消返回 cancel。位于 source 最内层，所
以缓存命中在离线时照常返回。
  - 删除：`DiomanOfflineException`（改为 `kind: offline`）。
  
> ## 6. envs
  
  只有 `setup`，不进链：按 `rules` 选第一个命中的配置写进 `client.dio.options`（写入前克隆，不与规则模板共享容器）。
  
  ## 7. 内核服务
  
  - **key**：算法沿用 0.7（深模式递归排序 Map、List 保序、`ignores`、`namespace`），由客户端级 `DiomanKeyOptions` 配置
。请求级 `call.key`：省略为自动；字符串为显式 key；`false` 表示不参与 cache/share。在 prepare 阶段之后惰性求值。
  - **取消**：`api.cancelAll(reason)` 返回本次中止的逻辑请求数；`api.pending`。
  - **debug**：`Dioman(debug: true)` 使用默认日志器；记录调用点、插件告警（例如请求体无法克隆、mock 缺少 `mockUrl`）。
  
  ## 8. TokenManager
  
  保留 `DiomanTokenManager` 接口（`accessToken`、`refreshToken`、`setTokens`、`clear` 等），删除 `canRefresh`：是否已判
过期的状态收进 auth 的实例内部。

import 'dart:async';
import 'dart:io';
import 'dart:typed_data';
import 'package:dioman/pipeline.dart';
import 'package:dio/dio.dart';
import 'package:test/test.dart';

/// 请求短超时，明显小于服务端停顿。
const _short = Duration(milliseconds: 150);

/// 请求长超时，覆盖正常分块或延迟响应。
const _long = Duration(milliseconds: 1200);

/// 底层宽松超时。
const _base = Duration(seconds: 10);

/// 测试路径。
const _path = '/timeout';

/// 请求方法。
const _get = 'GET';

/// 服务端响应文本。
const _payload = 'ok';

/// 大上传的一块大小。
const _chunkSize = 64 * 1024;

/// 大上传块数，足够阻塞回环缓冲区。
const _chunks = 1024;

/// 一次真实回环请求的资源与结果。
class _Loop {
  /// 绑定 Dio 和服务器资源。
  _Loop(this.server, this.release, this.api);

  /// 真实监听套接字。
  final HttpServer server;

  /// 释放挂起处理器。
  final Completer<void> release;

  /// 新管线客户端。
  final Dioman<DiomanResponse<dynamic>> api;

  /// 创建服务器，错误只隔离客户端关闭导致的服务端写失败。
  static Future<_Loop> start(String mode, DiomanTimeouts tier) async {
    final server = await HttpServer.bind(InternetAddress.loopbackIPv4, 0);
    final release = Completer<void>();
    server.listen((request) async {
      try {
        if (mode == 'headers') {
          request.response.bufferOutput = false;
          request.response.write(_payload);
          await request.response.flush();
          await release.future;
        } else if (mode == 'stall') {
          await release.future;
        } else if (mode == 'chunks') {
          request.response.bufferOutput = false;
          for (var i = 0; i < 4; i++) {
            request.response.write(_payload);
            await request.response.flush();
            await Future<void>.delayed(const Duration(milliseconds: 100));
          }
          await request.response.close();
        } else {
          if (mode == 'delay') {
            await Future<void>.delayed(const Duration(milliseconds: 400));
          } else {
            await request.drain<void>();
          }
          request.response.write(_payload);
          await request.response.close();
        }
      } on IOException {
        // 超时后客户端关闭套接字属于预期。
      }
    });
    final dio = Dio(BaseOptions(
        baseUrl: 'http://127.0.0.1:${server.port}',
        connectTimeout: _base,
        receiveTimeout: _base,
        sendTimeout: _base,
        responseType: ResponseType.plain));
    final api = Dioman(dio: dio, plugins: [
      DiomanTimeout(
          probe: () => NetworkQuality.poor,
          timeouts: {NetworkQuality.poor: tier})
    ]);
    final loop = _Loop(server, release, api);
    addTearDown(loop.close);
    return loop;
  }

  /// 释放全部真实 IO。
  Future<void> close() async {
    release.complete();
    api.dispose();
    api.dio.close(force: true);
    await server.close(force: true);
  }
}

/// 断言超时分类和实际经过的时长，排除即时失败冒充超时。
void _expectTimeout(
    DiomanResponse<dynamic> r, Stopwatch watch, Duration timeout) {
  expect(r.kind, DiomanKind.timeout);
  expect(watch.elapsed,
      greaterThanOrEqualTo(timeout - const Duration(milliseconds: 20)));
  expect(watch.elapsed, lessThan(timeout + const Duration(seconds: 2)));
}

/// 通过新入口验证 Dio 实际执行补丁超时。
void main() {
  for (final mode in ['headers', 'stall']) {
    test('[DiomanTimeouts.receive] $mode 停顿按插件值超时', () async {
      final loop =
          await _Loop.start(mode, const DiomanTimeouts(receive: _short));
      final watch = Stopwatch()..start();
      final r = await loop.api.send<Object?>(_get, _path);
      _expectTimeout(r, watch, _short);
      expect((r.error as DioException).type, DioExceptionType.receiveTimeout);
    });
  }
  test('[DiomanTimeouts.receive] 分块间隔小于超时即成功，总时长可超过超时', () async {
    const timeout = Duration(milliseconds: 250);
    final loop =
        await _Loop.start('chunks', const DiomanTimeouts(receive: timeout));
    final watch = Stopwatch()..start();
    final r = await loop.api.send<String>(_get, _path);
    expect(r.data, _payload * 4);
    expect(r.ok, isTrue);
    expect(watch.elapsed, greaterThan(timeout));
  });
  test('[DiomanTimeoutOptions.timeouts] 请求级收紧接收时限实际生效', () async {
    final loop =
        await _Loop.start('headers', const DiomanTimeouts(receive: _base));
    final watch = Stopwatch()..start();
    final r = await loop.api.send<Object?>(_get, _path,
        call: DiomanCall(options: {
          DiomanTimeout.option: const DiomanTimeoutOptions(
              timeouts: {NetworkQuality.poor: DiomanTimeouts(receive: _short)})
        }));
    _expectTimeout(r, watch, _short);
  });
  test('[DiomanTimeoutOptions.timeouts] 请求级放宽接收时限实际生效', () async {
    final loop =
        await _Loop.start('delay', const DiomanTimeouts(receive: _short));
    final failed = await loop.api.send<Object?>(_get, _path);
    expect(failed.kind, DiomanKind.timeout);
    final r = await loop.api.send<String>(_get, _path,
        call: DiomanCall(options: {
          DiomanTimeout.option: const DiomanTimeoutOptions(
              timeouts: {NetworkQuality.poor: DiomanTimeouts(receive: _long)})
        }));
    expect(r.ok, isTrue);
    expect(r.data, _payload);
  });
  test('[DiomanTimeouts.send] 发送缓冲阻塞时真实超时', () async {
    final loop = await _Loop.start('stall', const DiomanTimeouts(send: _short));
    final chunk = Uint8List(_chunkSize);
    final watch = Stopwatch()..start();
    final r = await loop.api.send<Object?>('POST', _path,
        data: Stream<List<int>>.fromIterable(
            Iterable.generate(_chunks, (_) => chunk)),
        call: const DiomanCall(
            headers: {Headers.contentLengthHeader: _chunkSize * _chunks}));
    _expectTimeout(r, watch, _short);
    expect((r.error as DioException).type, DioExceptionType.sendTimeout);
  });
  test('[DiomanTimeouts.send] 正常读取小请求体时成功', () async {
    final loop = await _Loop.start('read', const DiomanTimeouts(send: _short));
    final r = await loop.api.send<String>('POST', _path, data: [1, 2, 3]);
    expect(r.ok, isTrue);
    expect(r.data, _payload);
  });
}

  ## 7. 插件接口
  
  ```dart
> abstract class DiomanPlugin<R extends DiomanResponse> {
    String get name;                              // 内置插件以 dioman: 开头
    DiomanStage get stage => DiomanStage.attempt; // 自定义插件的默认阶段
    DiomanEnforce get enforce => DiomanEnforce.post;
    /// 每个客户端调用一次；状态建在返回的实例里。只有 setup、没有 handle 的插件不进链。
    DiomanPluginInstance<R> setup(Dioman<R> client);
  }
> abstract class DiomanPluginInstance<R extends DiomanResponse> {
    Future<R> handle(DiomanContext<R> ctx, DiomanNext<R> next);
    void dispose() {}
  }
  ```
  
  约定：
  - **状态必须建在 `setup` 返回的实例里**，不能放在插件对象上。插件对象只是配置（可以是 const），同一个插件对象可以装进
多个客户端，各自一份状态。
  - 插件暴露给使用者的操作（例如 cache 的 `clear`），通过 `api.plugin<DiomanCacheApi>()` 取得。
  - 用户回调分两类：**通知类**（loading、notify、onAccessExpired、onAccessDenied、onStateChange、onConnectivityChanged
）抛错会被隔离，并输出到 `debugPrint` 级别的错误日志；**判定类**（shouldRetry、delay、onFailure、isProtected、fallbackW
hen、shouldCache、shouldTrip、keyBuilder）抛错按 bug 处理。`onRefresh` 例外：它抛错算刷新失败。

  ```dart
            Object Function(R r)? reject, Object? debug /* bool | DiomanLogger */,
            DiomanClock clock = const DiomanClock.system()});
    Dio get dio;
    Future<T> get<T>(String path, {Map<String, dynamic>? query, Object? data, DiomanCall? call});
    // post / put / patch / delete / head 同形；request<T>(method, path, …)
    Future<DiomanResponse<T>> send<T>(String method, String path, {…});
    void use(DiomanPlugin<R> plugin);
    bool eject(String name);
    A? plugin<A>();                       // 取插件暴露的操作
    List<(DiomanStage, String)> inspect();

    int get pending;
>   Dioman<R> extend({BaseOptions? options, List<DiomanPlugin<R>>? plugins});

  }
  ```
  
  - `DiomanCall`：调用级配置，包含 headers、超时、使用者的 `cancelToken`、`key`、`silent`、以及各插件的请求级选项（以插
件导出的选项键为键）。
> - `extend()` 克隆 Dio 的 BaseOptions，默认继承插件列表并各自重新 `setup`，状态不共享。
  - `dispose()`：先 `cancelAll()`，再按注册的逆序调用各插件实例的 `dispose`（尽力执行完，最后抛出第一个异常）；之后拒绝
新的请求和 `use`。
  
  ## 8.1 B1 终审裁决（2026-10-29）
  
  - **链内返回类型为 `DiomanResponse<dynamic>`**，不是使用者的子类 `R`：使用者工厂自身抛错时，内核只能用内置工厂造基础

/// 有效 Dio 配置的不可变请求视图；入口合并一次后构造。
class DiomanRequest {
  /// 从 [options] 拷贝有效配置；[call] 保存调用选项，例如 `DiomanRequest(ro)`。
  DiomanRequest(RequestOptions options, {DiomanCall call = const DiomanCall()})
      : _options = options.copyWith(),
        data = options.data,
        headers = Map.unmodifiable(options.headers),
        query = Map.unmodifiable(options.queryParameters),
        extra = Map.unmodifiable(options.extra),
        call = DiomanCall(
            key: call.key,
            envelope: call.envelope,
            silent: call.silent,
            options: Map.unmodifiable(call.options));

  /// 完整有效配置的私有副本，永不交给 Dio 修改。
  final RequestOptions _options;

  /// 请求体引用；不透明载荷由传输按需克隆，插件通过补丁替换。
  final Object? data;

  /// 不可修改的请求头。
  final Map<String, dynamic> headers;

  /// 不可修改的查询参数。
  final Map<String, dynamic> query;

  /// 不可修改的用户扩展字段。
  final Map<String, dynamic> extra;

  /// 调用配置。
  final DiomanCall call;

  /// 有效路径。
  String get path => _options.path;

  /// 有效方法。
  String get method => _options.method;

  /// 有效基础地址。
  String get baseUrl => _options.baseUrl;

  /// 有效 URI。
  Uri get uri => toRequestOptions().uri;

  /// 连接超时。
  Duration? get connectTimeout => _options.connectTimeout;

  /// 发送超时。
  Duration? get sendTimeout => _options.sendTimeout;

  /// 接收超时。
  Duration? get receiveTimeout => _options.receiveTimeout;

  /// 当前传输令牌；未指定时内核使用逻辑请求令牌。
  CancelToken? get cancelToken => _options.cancelToken;

  /// 通知静默标记。
  bool get silent => call.silent;

  /// 返回可独立修改的有效 Dio 配置，例如 `dio.fetch(r.toRequestOptions())`。
  RequestOptions toRequestOptions() => _options.copyWith(
      headers: Map.from(headers),
      queryParameters: Map.from(query),
      extra: Map.from(extra))
    ..data = data;

  /// 合并 [patch] 并返回独立视图；不修改基准请求。
  DiomanRequest patch(RequestPatch patch) {
    final o = toRequestOptions();
    final h = LinkedHashMap<String, dynamic>.from(o.headers);
    for (final key in patch.removeHeaders) {
      h.removeWhere((k, _) => k.toLowerCase() == key.toLowerCase());
    }
    for (final e in patch.headers.entries) {
      h.removeWhere((k, _) => k.toLowerCase() == e.key.toLowerCase());
      h[e.key] = e.value;
    }
    o.headers = h;
    o.queryParameters.removeWhere((k, _) => patch.removeQuery.contains(k));
    o.queryParameters.addAll(patch.query);
    o.extra.addAll(patch.extra);
    if (!identical(patch.data, _absent)) o.data = patch.data;
    if (patch.path != null) o.path = patch.path!;
    if (patch.method != null) o.method = patch.method!;
    if (!identical(patch.connectTimeout, _absent)) {
      o.connectTimeout = patch.connectTimeout as Duration?;
    }
    if (!identical(patch.sendTimeout, _absent)) {
      o.sendTimeout = patch.sendTimeout as Duration?;
    }
    if (!identical(patch.receiveTimeout, _absent)) {
      o.receiveTimeout = patch.receiveTimeout as Duration?;
    }
    if (!identical(patch.cancelToken, _absent)) {
      o.cancelToken = patch.cancelToken as CancelToken?;
    }
    return DiomanRequest(o, call: call);
  }
}
