/// 缓存管理能力；通过 `api.plugin<DiomanCacheApi>()` 获取。
abstract interface class DiomanCacheApi {
  /// 删除 [key] 并作废该键在途写入；等待持久化删除完成。
  Future<void> remove(String key);

  /// 清空缓存并立即作废所有在途写入；返回持久化清理完成信号。
  Future<void> clear();

  /// 在途屏障数，只读诊断；所有请求结束后为零。
  int get pendingWrites;

  /// 内存物理条目数，包含尚未惰性过期的条目。
  int get entryCount;
}

/// 按请求开启的缓存；命中不触网，业务失败默认不存储。
class DiomanCache<R extends DiomanResponse<dynamic>> extends DiomanPlugin<R> {
  /// 默认 memo、一分钟、浅克隆、GET/HEAD；例如选项 `{DiomanCache.option: true}`。
  const DiomanCache(
      {this.persist,
      this.maxEntries = 500,
      this.cachePolicy = DiomanCachePolicy.memo,
      this.expires = const Duration(minutes: 1),
      this.clone = DiomanClonePolicy.shallow,
      this.clock,
      this.shouldCache,
      this.shouldStore,
      this.enabled = true});

  /// 内存 LRU 上限，非正数不限；请求级不支持覆盖。
  final int maxEntries;

  /// 可选持久化存储；persist/both 策略要求提供。
  final DiomanCachePersist? persist;

  /// 默认存储层。
  final DiomanCachePolicy cachePolicy;

  /// 默认有效期。
  final Duration expires;

  /// 默认克隆策略或自定义函数。
  final Object clone;

  /// 默认请求判定，缺省 GET/HEAD。
  final bool Function(DiomanRequest)? shouldCache;

  /// 默认写入判定，缺省 r.ok。
  final bool Function(DiomanResponse<dynamic>)? shouldStore;

  /// 可选 TTL 时钟；默认客户端 DiomanClock，请求级不支持覆盖。
  final DiomanClock? clock;

  /// 安装期开关。
  @override
  final bool enabled;

  /// 稳定名称。
  static const pluginName = 'dioman:cache';

  /// 请求级 bool、Duration 或配置键。
  static final option = DiomanOptionKey<DiomanCacheOptions>(pluginName);

  /// 名称。
  @override
  String get name => pluginName;

  /// 结果来源阶段。
  @override
  DiomanStage get stage => DiomanStage.source;

  /// 状态按客户端隔离，时钟统一使用客户端时钟。
  @override
  DiomanPluginInstance<R> setup(Dioman<R> client) => _Cache<R>(
      persist,
      maxEntries,
      clock ?? client.clock,
      DiomanCacheOptions(
          cachePolicy: cachePolicy,
          expires: expires,
          clone: clone,
          shouldCache: shouldCache,
          shouldStore: shouldStore));

/// 内存缓存条目。
class _Entry {
  /// 保存响应及绝对到期时刻。
  _Entry(this.response, this.expiresAt);

  /// 来源响应。
  final DiomanResponse<dynamic> response;

  /// epoch 毫秒。
  final int expiresAt;

/// 在途期间的已写入顺序屏障。
class _Barrier {
  /// 当前在途数量。
  int pending = 0;

  /// 已发布的最大序号。
  int latest = 0;
}

/// 每客户端内存和串行持久化通道。
class _Cache<R extends DiomanResponse<dynamic>> extends DiomanPluginInstance<R>
    implements DiomanCacheApi {
  /// 保存存储与配置。
  _Cache(this.persist, this.maxEntries, this.clock, this.base);

  /// TTL 时钟。
  final DiomanClock clock;

  /// 内存容量。
  final int maxEntries;

  /// 外部持久化后端。
  final DiomanCachePersist? persist;

  /// 基础选项。
  final DiomanCacheOptions base;

  /// 内存条目。
  final entries = <String, _Entry>{};

  /// 只在在途期间存在的屏障。
  final barriers = <String, _Barrier>{};

  /// 单调请求序号。
  int sequence = 0;

  /// 最近 clear 的失效下界。
  int floor = 0;

  /// 所有落盘变更串行，clear 不会被旧 write 越过。
  Future<void> tail = Future<void>.value();

  /// 数据字段名，兼容旧存储。
  static const dataField = 'data';

  /// 到期字段名。
  static const expiresField = 'expiresAt';

  /// 新格式保存归一化元信息，旧格式仍按原始 HTTP 200 读取。
  static const resultField = 'result';

  /// 结果类别字段。
  static const kindField = 'kind';

  /// 逻辑状态字段。
  static const statusField = 'status';

  /// 原始 HTTP 状态字段。
  static const httpField = 'httpStatus';

  /// 业务码字段。
  static const codeField = 'code';

  /// 消息字段。
  static const messageField = 'message';

  /// 细分原因字段。
  static const reasonField = 'reason';

  /// 从新格式恢复归一化结果，老格式继续交给工厂识别信封。
  DiomanResponse<dynamic> restore(DiomanContext<R> ctx, Map saved) {
    final meta = saved[resultField];
    if (meta is! Map) {
      return ctx.factory.normalize(ctx.request,
          raw: Response<dynamic>(
              requestOptions: ctx.request.toRequestOptions(),
              statusCode: 200,
              data: saved[dataField]));
    }
    final kind = DiomanKind.values.byName(meta[kindField] as String);
    final init = DiomanResponse<dynamic>(
        ok: kind == DiomanKind.ok,
        kind: kind,
        status: meta[statusField] as int,
        httpStatus: meta[httpField] as int,
        code: meta[codeField],
        message: meta[messageField] as String,
        reason: meta[reasonField] as String?,
        data: meta[dataField],
        request: ctx.request);
    return ctx.factory.derive(init, init.data);
  }

  /// 在途屏障诊断。
  @override
  int get pendingWrites => barriers.length;

  /// 内存条目诊断。
  @override
  int get entryCount => entries.length;

  /// 写入并按最近使用顺序淘汰，屏障不随淘汰删除。
  void put(String key, _Entry entry) {
    entries.remove(key);
    entries[key] = entry;
    if (maxEntries > 0) {
      while (entries.length > maxEntries) {
        entries.remove(entries.keys.first);
      }
    }
  }

  /// 入队持久化变更；异常交给调用方，同时保持通道可继续使用。
  Future<void> enqueue(Future<void> Function() action) {
    final pending = tail.then((_) => action());
    tail = pending.then((_) {}, onError: (Object _, StackTrace __) {});
    return pending;
  }

  /// 删除也抬高本键屏障，阻止旧响应复活。
  @override
  Future<void> remove(String key) {
    entries.remove(key);
    barriers[key]?.latest = ++sequence;
    return persist == null
        ? Future<void>.value()
        : enqueue(() => persist!.remove(key));
  }

  /// 同步失效，随后串行清理后端。
  @override
  Future<void> clear() {
    floor = sequence;
    entries.clear();
    return persist == null
        ? Future<void>.value()
        : enqueue(() => persist!.erase());
  }

  /// 递归复制 Map/List；其余载荷按引用保留。
  Object? deep(Object? data) {
    if (data is Map) return data.map((k, v) => MapEntry(k, deep(v)));
    if (data is List) return data.map(deep).toList();
    return data;
  }

  /// 数据副本经扩展工厂创建，保留所有响应元信息。
  DiomanResponse<dynamic> clone(
      DiomanContext<R> ctx, DiomanResponse<dynamic> r, Object strategy) {
    if (strategy == DiomanClonePolicy.none) return r;
    final data = r.data;
    Object? copied;
    if (strategy is Object? Function(Object?)) {
      copied = strategy(data);
    } else if (strategy == DiomanClonePolicy.deep) {
      copied = deep(data);
    } else if (strategy == DiomanClonePolicy.shallow) {
      copied = data is Map
          ? Map.of(data)
          : data is List
              ? List.of(data)
              : data;
    } else {
      throw ArgumentError.value(strategy, 'clone');
    }
    return ctx.factory.derive(r, copied);
  }

  /// 查缓存、登记屏障、只写入仍新鲜的结果，并在 finally 回收。

  @override
  int get activeEntries =>

  /// 仅移除仍属于本轮的条目。
  void detach(Map<String, _Shared> table, String key, _Shared entry) {
    if (identical(table[key], entry)) table.remove(key);
  }

  /// 结果发布前先摘表，外层 retry 可立即重新入表。
  void finish(Map<String, _Shared> table, String key, _Shared entry,

    for (final stage in [
      request.connectTimeout,
      request.receiveTimeout,
      if (request.data != null) request.sendTimeout
    ]) {
      if (stage == null || stage <= Duration.zero) return null;
      total += stage;
    }
    return total;
  }

      DiomanShareOptions o,

      String key,
      _Shared entry) async {

          key,

    } finally {

  }

  /// end/race 的底层请求各自独立，订阅者离开不会取消其它传输。
  Future<void> compete(

    }
  }

/// 客户端专属 FIFO 队列。
class _Offline<R extends DiomanResponse<dynamic>>
    extends DiomanPluginInstance<R> implements DiomanOfflineApi {
  /// 订阅宿主连通流，流错误归通知类隔离。
  _Offline(this.config, Dioman<R> client) {

      if (online) flush();

  late final DiomanOfflineOptions base;

  /// 宿主订阅。
  late final StreamSubscription<bool> subscription;

  /// 按入队顺序排列。
  final queue = ListQueue<_Queued>();

  /// 是否已释放，dispose 的原因优先于客户端同时发出的取消。
  bool disposed = false;

  /// 当前排队数量。
  @override
  int get pending => queue.length;

  /// 清队列后发信号，保证重入安全。
  void drain(String? reason) {
    final entries = queue.toList();
    queue.clear();
    for (final entry in entries) {
      entry.ready.complete(reason);
    }
  }

  /// 手动或流通知恢复。
  @override
  void flush() => drain(null);

  /// 离线等待统一使用 ctx.wait，finally 摘掉队列条目。

            enabled: b.enabled,
            shouldQueue: b.shouldQueue ?? a.shouldQueue,
            maxQueueSize: b.maxQueueSize ?? a.maxQueueSize,
            maxWait: b.maxWait ?? a.maxWait));
    if (o == null ||
        o.enabled == false ||
        config.isOnline() ||
        !(o.shouldQueue?.call(ctx.request) ?? true)) {
      return next();
    }
    if (disposed) {
      ctx.set(cancellationFailure, _disposedFailure);
      return ctx.fail(_disposedFailure);
    }
    final cancel = ctx.cancelled();
    if (cancel != null) return cancel;
    final max = o.maxQueueSize!;
    if (max > 0) {
      while (queue.length >= max) {
        queue.removeFirst().ready.complete(_queueFullReason);
      }
    }
    final entry = _Queued();
    queue.add(entry);
    try {
      final ready = await ctx.wait(
          entry.ready.future.then((reason) => (reason,)),
          timeout: o.maxWait);
      if (disposed) {
        ctx.set(cancellationFailure, _disposedFailure);
        return ctx.fail(_disposedFailure);
      }
      final cancel = ctx.cancelled();
      if (cancel != null) return cancel;
      if (ready == null) {

            kind: DiomanKind.offline, reason: _timeoutReason));
      }
      final reason = ready.$1;
      return reason == null
          ? await next()
          : ctx.fail(DiomanFailure(kind: DiomanKind.offline, reason: reason));
    } finally {
      queue.remove(entry);
    }
  }

  @override
  void dispose() {
    disposed = true;
    unawaited(subscription.cancel());
    drain(_disposedReason);
  }

final class StateKey<T> {
  /// 创建键，例如 `final attempt = StateKey<int>('attempt')`。

/// 可解除目标的 Dio 取消桥；Dio Future 自身没有退订接口。
class DiomanCancelLink {
  /// 将 [source] 联动到 [target]，已取消时同步传递。
  DiomanCancelLink(CancelToken source, void Function(Object?) target)
      : _target = target {
    if (source.isCancelled) {
      target(source.cancelError);
    } else {
      source.whenCancel.then(_deliver);
    }
  }

  /// 可清除的订阅目标，清理后不再引用请求状态。
  void Function(Object?)? _target;

  /// 转发用户取消。
  void _deliver(Object? reason) => _target?.call(reason);

  /// 解除逻辑订阅；迟到的取消不再触及请求。
  void dispose() => _target = null;
}

/// 一个逻辑请求的共享状态和取消订阅登记。
class DiomanContextState {
  /// 创建状态并监听内核令牌；测试也可独立构造上下文。
  DiomanContextState(this.token) {
    _link = DiomanCancelLink(token, (_) => _notify());
  }

  /// 内核逻辑取消令牌。
  final CancelToken token;

  /// Dio 到内核订阅表的桥。
  late final DiomanCancelLink _link;

  /// 惰性分配的类型化值。
  Map<Object, Object?>? values;

  /// 按请求身份缓存 key。
  final keys = Map<DiomanRequest, String?>.identity();

  /// 等待取消的订阅表。
  final listeners = <void Function(), CancelToken>{};

  /// 是否需要克隆发送体。
  bool replayable = false;

  /// 脱离首发调用方但仍在执行的共享传输数。
  int _retained = 0;

  /// 逻辑调用方是否已完成。
  bool _released = false;

  /// 保持共享传输使用的状态；每次调用必须对应一次 release。
  void retain() => _retained++;

  /// 共享传输完成后释放；最后一个持有者负责最终清理。
  void release() {
    _retained--;
    if (_released && _retained == 0) dispose();
  }

  /// 通知副本，允许回调在执行中退订。
  void _notify() {
    for (final entry in listeners.entries.toList()) {
      if (identical(entry.value, token)) entry.key();
    }
  }

  /// 同步取消内核请求并通知等待者。
  void cancel([Object? reason]) {
    token.cancel(reason);
    _notify();
  }

  /// 释放桥并结束剩余等待。
  void dispose() {
    _released = true;
    if (_retained > 0) return;
    _link.dispose();
    for (final listener in listeners.keys.toList()) {
      listener();
    }
    listeners.clear();
  }
}

/// 每层的固定形状上下文；派生请求共享逻辑状态。
class DiomanContext<R extends DiomanResponse<dynamic>> {
  /// 注入内核服务；例如测试可直接构造而不发起请求。
  DiomanContext(
      {required this.id,
      required this.client,
      required this.request,
      required this.state,
      required this.factory,
      required this.keyService,
      required this.clock,
      required this.log});

  /// 请求序号。
  final int id;

  /// 所属客户端。
  final Dioman<R> client;

  /// 本层请求视图。
  final DiomanRequest request;

  /// 共享状态。
  final DiomanContextState state;

  /// 结果服务。
  final DiomanResponseFactory<R> factory;

  /// 构键服务。
  final DiomanKeyService keyService;

  /// 注入时钟。
  final DiomanClock clock;

  /// 受 debug 开关控制的日志器。
  final DiomanLogger log;

  /// 当前传输的取消令牌，补丁可以覆盖。
  CancelToken get cancelToken => request.cancelToken ?? state.token;

  /// 逻辑请求或当前传输是否取消。
  bool get isCancelled => cancelToken.isCancelled;

  /// 当前逻辑请求残留的等待订阅数。
  int get cancellationSubscriptions => state.listeners.length;

  /// 是否保护重放请求体。
  bool get replayable => state.replayable;

  /// 惰性构键并按请求身份缓存；false 选项返回 null。
  String? key() {
    if (!state.keys.containsKey(request)) {
      state.keys[request] =
          keyService.build(request.toRequestOptions(), request.call.key);
    }
    return state.keys[request];
  }

  /// 读取 [key] 的类型化值，未设置返回 null。
  T? get<T>(StateKey<T> key) => state.values?[key] as T?;

  /// 写入 [key] 的类型化 [value]，各层共享。
  void set<T>(StateKey<T> key, T value) => (state.values ??= {})[key] = value;

  /// 经工厂生成结果；坏扩展工厂回退基础 internal，以保证 next 不抛错。
  DiomanResponse<dynamic> fail(DiomanFailure failure) =>
      factory.fail(request, failure);

  /// 声明可能重放；后续每次发送均独立克隆请求体。
  void markReplayable() => state.replayable = true;

  /// 已取消则造 cancel 结果，否则返回 null；每次 await 后应调用。
  DiomanResponse<dynamic>? cancelled() => isCancelled
      ? fail(get(cancellationFailure) ??
          const DiomanFailure(kind: DiomanKind.cancel))
      : null;

  /// 派生请求视图，保留所有逻辑服务与共享状态。
  DiomanContext<R> derive(RequestPatch patch) => DiomanContext(
      id: id,
      client: client,
      request: request.patch(patch),
      state: state,
      factory: factory,
      keyService: keyService,
      clock: clock,
      log: log);

  /// 等待 [pending]、取消或 [timeout]；后两者返回 null，结束后无订阅及计时器。
  /// 已取消立即返回；迟到的值或异常被消费且不会改变结局。
  Future<T?> wait<T>(Future<T> pending, {Duration? timeout}) {
    final result = Completer<T?>();
    Timer? timer;
    DiomanCancelLink? localLink;
    late void Function() cancel;
    void clean() {
      timer?.cancel();
      localLink?.dispose();
      state.listeners.remove(cancel);
    }

    void finish(T? value) {
      if (result.isCompleted) return;
      clean();
      result.complete(value);
    }

    cancel = () => finish(null);
    // 即使已经取消，也消费输入的迟到异常。
    pending.then((value) => finish(value),
        onError: (Object error, StackTrace stack) {
      if (result.isCompleted) return;
      clean();
      result.completeError(error, stack);
    });
    if (isCancelled) {
      finish(null);
      return result.future;
    }
    state.listeners[cancel] = cancelToken;
    if (!identical(cancelToken, state.token)) {
      localLink = DiomanCancelLink(cancelToken, (_) => cancel());
    }
    if (timeout != null) timer = clock.createTimer(timeout, cancel);
    return result.future;
  }
}

typedef DiomanRunner<R extends DiomanResponse<dynamic>>

/// 已安装的配置及实例。
class DiomanRegistration<R extends DiomanResponse<dynamic>> {
  /// 保存配置与客户端实例。
  const DiomanRegistration(this.plugin, this.instance);

  /// 插件配置。
  final DiomanPlugin<R> plugin;

  /// setup 返回的独立状态。
  final DiomanPluginInstance<R>? instance;

/// 按名称确定位置；自定义插件位于同阶段内置插件前后。
(int, int) _position(DiomanPlugin plugin) {
  for (final entry in diomanBuiltinStages.entries) {
    final index = entry.value.indexOf(plugin.name);
    if (index >= 0) return (entry.key.index, index);
  }
  if (plugin.name.startsWith('dioman:')) {
    throw ArgumentError.value(plugin.name, _nameArgument, '保留前缀');
  }
  return (plugin.stage.index, plugin.enforce == DiomanEnforce.pre ? -1 : 100);
}

/// 在 setup 前验证重名和保留前缀，拒绝非法配置且不产生副作用。
void validateDiomanPlugins(Iterable<DiomanPlugin> plugins) {
  final names = <String>{};
  for (final plugin in plugins) {
    if (!names.add(plugin.name)) {
      throw ArgumentError.value(plugin.name, _nameArgument, '插件重名');
    }
    if (!diomanSetupPlugins.contains(plugin.name)) _position(plugin);
  }
}

/// 稳定排序已启用且有 handle 的插件；不修改输入。
List<DiomanRegistration<R>>
    sortDiomanPipeline<R extends DiomanResponse<dynamic>>(
        List<DiomanRegistration<R>> registrations) {
  final indexed =
      registrations.indexed.where((e) => e.$2.instance != null).toList();
  indexed.sort((a, b) {
    final x = _position(a.$2.plugin);
    final y = _position(b.$2.plugin);
    final stage = x.$1.compareTo(y.$1);
    if (stage != 0) return stage;
    final order = x.$2.compareTo(y.$2);
    return order == 0 ? a.$1.compareTo(b.$1) : order;
  });
  return indexed.map((e) => e.$2).toList();
}

/// 从内到外编译一次；每层含一个异常边界，无补丁复用上下文。
DiomanRunner<R> compileDiomanPipeline<R extends DiomanResponse<dynamic>>(
    List<DiomanRegistration<R>> layers, DiomanRunner<R> transport) {
  var run = transport;
  for (final layer in layers.reversed) {
    final inner = run;
    run = (ctx) async {
      try {
        return await layer.instance!.handle(ctx, ([patch]) async {
          DiomanContext<R> target;
          try {
            target = patch == null ? ctx : ctx.derive(patch);
          } catch (e, s) {
            return ctx.fail(DiomanFailure(
                kind: DiomanKind.internal, error: e, errorStack: s));
          }
          return inner(target);
        });
      } catch (e, s) {
        return ctx.fail(
            DiomanFailure(kind: DiomanKind.internal, error: e, errorStack: s));
      }
    };
  }
  return run;
}

  Dioman(

  final bool _debug;

  Future<DiomanResponse<T>> send<T>(String method, String path,

        await _dispatch(method, path, query: query, data: data, call: call);
    if (result is DiomanResponse<T>) return result;
    return result.copy<T>();

  /// 执行任意 [method] 并返回数据；可预期失败经 reject 后抛出。
  Future<T> request<T>(String method, String path,

    if (!result.ok) throw (reject?.call(result as R) ?? result);
    return result.data as T;
  }

  Future<T> get<T>(String path,

      request<T>('GET', path, query: query, data: data, call: call);

  /// 发送 POST 并返回数据；参数与 request 相同，例如 `api.post<Object?>('/pets')`。
  Future<T> post<T>(String path,

  /// 发送 PUT 并返回数据；参数与 request 相同，例如 `api.put<Object?>('/pets')`。
  Future<T> put<T>(String path,

## TL;DR

- 12 个内置插件：outer 阶段 loading、notify；prepare 阶段 repath、filter、timeout；attempt 阶段 retry、breaker；credential 阶段 auth；source 阶段 mock、share、cache、offline。envs 只有 `setup`，不进链。
- key、cancel、normalize、log 降为内核能力（见[核心架构](pipeline-architecture.md)）。
- 现有功能全部保留，只改写法；行为变化集中列在 §9，CHANGELOG 和迁移指南按它写。

## 0. 插件约定

1. 工厂即类：`DiomanRetry(…)` 等，`name` 为 `dioman:<名字>`。插件对象只放配置，可以是 const；状态建在 `setup` 返回的实例里。
2. **插件级 `enabled: false`**：内核在编译链时直接跳过，不调 `setup`，`inspect()` 不列出。
3. **请求级选项统一四种写法**，由 `resolveOption` 一个函数解析：省略、`true`（用插件级配置）、`false`（本次关闭）、插件自己的选项对象（字段为 null 的沿用插件级）。部分插件另有一种简写（retry 的整数 = `max`，cache 的 `Duration` = `expires`，share 的策略枚举）。不认识的类型按 bug 抛 `ArgumentError`。插件级配置在 `setup` 时预合并；请求级 `true` 或省略时复用同一份。
4. **装上即生效**，只有改变数据新鲜度的 **cache、share 要在请求级开启**（optIn）。
5. **改请求只用 `next(patch)`**，不改 `ctx.request`。
6. 用户回调抛错：通知类隔离并输出错误日志；判定类按 bug 处理；`onRefresh` 抛错算刷新失败（见核心架构 §7）。
7. **每个 await 之后用 `ctx.cancelled()` 重查取消**；等待一律用 `ctx.wait`。
8. 会重发请求的插件（retry、auth、mock、share 的 retry 策略）在第一次 `next()` 之前调用 `ctx.markReplayable()`。

## 1. outer

### loading
- 选项：`onChanged(bool visible)` 必填；`delay`（0→1 之后延迟显示，期间回落就不显示）、`delayClose`（1→0 之后延迟隐藏，期间有新请求就不隐藏），默认都是 `Duration.zero`。请求级：`false`、`true`、`DiomanLoadingOptions`。
- 模型：计数加一；`try { return await next(); } finally { 计数减一 }`。一个逻辑请求恰好一对，重试和重放都包在里面。按回调身份分开计数：请求级换了回调的请求，计入自己那个回调。
- `dispose`：取消未触发的计时器；对已显示的回调立即发一次 `false`；在途请求不再通知。
- 内核依赖：outer 阶段位置、`DiomanClock`。

### notify
- 选项：`notify(String message, R r)` 必填；`stringify(R r, DiomanContext ctx)`，默认在 `r.ok`、`kind == cancel`、`ctx.request.silent` 时返回空串，否则返回 `r.message`，为空时返回 `'<kind> <status>'`。空串表示不通知。请求级只有开关。
- 模型：`final r = await next(); 通知; return r;`。每个逻辑请求最多一次，天然只在最终失败时通知，刷新成功不通知。
- 内核依赖：outer 阶段、`request.silent`。

## 2. prepare

### repath
- 选项不变：`pattern`（默认匹配 `{id}`、`:id`、`[id]`）、`removeKey`（默认 true）。只处理路径部分，先找 query（含 BaseOptions 的 queryParameters）再找 Map 形式的 data；值按路径段编码；同名占位符多次出现全部替换。算出新的 path/query/data 后 `next(patch)`；没有变化直接 `next()`。

### filter
- 选项：`predicate(key, value)`，默认剔除 null、NaN、空白字符串，保留 0 与 false；`ignoreKeys`、`ignoreValues`。剥 query 和 Map 形式的 data 的顶层字段；FormData、Stream 原样。没有可剔除的字段时不打补丁。
- 变化：**装上即生效**（0.7 也是），默认谓词新增剔除 NaN（testkit 中立语义，且 `jsonEncode(double.nan)` 会抛错）。

### timeout
- 选项不变：`timeouts`（按 `NetworkQuality` 档位的 connect/receive/send）、`probe`（返回当前网络档位）。请求级：`false`、`DiomanTimeoutOptions`。
- 模型：读出本次档位对应的超时，`next(patch)` 写入请求的超时字段。只改配置，真正计时由 Dio 执行（假设 H5，已在本地回环核对）。

## 3. attempt

### retry
- 选项沿用 0.7：`max`、`methods`（硬否决）、`statusCodes`（默认 408/429/500/502/503/504，读**逻辑状态** `r.status`）、`shouldRetry(R r, int attempt)`、`delay`、`jitter`、`delayMax`、`respectRetryAfter`（默认 true，命中 `afterStatusCodes` 默认 413/429/503 时按 `Retry-After` 的秒数或 HTTP 日期等待，上限 `retryAfterMax`，不叠 jitter、不受 delayMax 限制）、`onRetry(attempt)`。请求级：整数 = max、`false`、`true`、`DiomanRetryOptions`。
- 判定顺序：请求级关闭 → 方法不在白名单 → `kind` 属于 cancel、auth、internal、parse、breaker、offline 时不重试 → `shouldRetry` 返回布尔就采用 → network、timeout 重试 → `statusCodes` 包含 `r.status` 时重试。
- 模型：`markReplayable()`；`var r = await next(); for (attempt = 1..max) { 判定; 等待 ctx.wait(delay); 取消则返回; r = await next(); } return r;`。次数用完返回最后一次结果，由落定决定抛不抛。每次尝试都会穿过 breaker、auth 和 source 阶段。
- 删除：重放交接、认领判断（`_notAdmitted`）、`share`/`cancel` 兼容 setter、`breaker` 字段。

### breaker
- 选项沿用 0.7：`failureThreshold`、`resetDuration`、`halfOpenMaxCalls`、`shouldTrip(R r)`（默认：`kind` 是 network/timeout，或逻辑状态 ≥500 或 429 时计失败；cancel、internal 不计）、`keyBuilder(request)`（默认 `METHOD:path`）、`onStateChange`、`idleTtl`、`maxBuckets`、`clock`。
- 模型：先申请放行；熔断打开或半开名额已满时，直接返回 `ctx.fail(kind: breaker, reason: 'open')`。放行后 `r = await next()`；按 `shouldTrip` 记一次成或败；cancel 只归还名额、不计成败。在 retry 之内，所以每次尝试都会计数；熔断打开时 retry 遇到 `kind: breaker` 不再重试，不需要接线。
- 删除：`admit/recordResponse/recordError` 公开方法（原本只为 retry 接线而存在）、`DiomanBreakerOpenException`（改为 `kind: breaker`）。

## 4. credential：auth

- 选项沿用 0.7：`tokenManager`、`onRefresh`、`onAccessExpired` 必填；`onAccessDenied`（缺省回退 expired）；`onFailure(R r, tokenManager)`，默认见下；`isProtected(request)`；`ready(tokenManager, request)` 返回 `RequestPatch`，默认写 `Authorization: Bearer <token>`（`headerKey`、`buildHeader` 保留为便捷写法）；`expiresAt`、`refreshLeeway`、`clock` 用于主动刷新。请求级：`false`、`DiomanAuthOptions`。
- 默认失败判定（读逻辑状态，所以 business 风格里的业务 401 自动适用）：`r.status == 401` 时，本地没有 token → expired；**本次注入的 token 不等于当前 token → replay**（说明已被并发刷新）；否则 refresh。`403` → deny；其余 → others。本次注入的 token 记在状态键里，不去解析请求头。
- 模型：
  1. 不受保护，直接 `return next()`；`markReplayable()`。
  2. 如果正在刷新，`ctx.wait` 等刷新结果；取消就返回；刷新失败返回 `ctx.fail(kind: auth, reason: 'expired')`。
  3. 循环：本地没有 token → 通知 denied，返回 `reason: 'unauthenticated'`；即将过期且配了 `expiresAt` → 先主动刷新；注入 token 并 `r = await next(patch)`；ok、internal、cancel 直接返回；`action = await onFailure(r)`，回来**先查取消**；refresh/replay：已经重放过一次就按 expired 处理（防止循环），否则 refresh 时加入或发起单飞刷新，然后再循环一次；deny → 通知并返回 `reason: 'denied'`；expired → 清会话、通知、返回 `reason: 'expired'`；others → 原样返回。
- **会话判定**沿用 2026-10-28 的修复：按失败请求实际携带的 token 判断属于哪个会话，旧请求的失败不清新会话，每个会话只通知一次过期；单飞刷新以 `Future(() => onRefresh(…))` 强制异步发起，同步抛错也不会卡死刷新窗口。
- **重放前重查本地凭证**：刷新成功后、重放之前使用者登出，就不再重放（axp B3 抓到的缺陷）。
- 删除：重放交接、认领结算、`DiomanAuth.dio`/`share` setter。

## 5. source：mock → share → cache → offline

### mock
- 选项：`enabled`（默认 true，装上即生效）、`routes`（`METHOD:path` → 处理函数，内联 mock）、`mockUrl`（转发到 mock 服务）、`fallbackWhen(R r)`，默认 `r.httpStatus == 404 || r.kind == network`。请求级：`false`、`true`、`DiomanMockOptions`。
- 模型：命中内联路由 → 用处理函数的结果经工厂造 `R`，直接返回。配了 `mockUrl` → `markReplayable()`，`probe = await next(patch: 改写到 mockUrl，并带"不走 cache、不参与 share"的状态标记)`；`fallbackWhen(probe)` 为真时再 `next()` 打真实接口。
- 变化：从传输层 adapter（2026-10-28）回到 source 阶段，`DiomanMockAdapter` 删除，cache 的 `x-dioman-mock` 规则删除。具备 testkit 的 `mock-in-chain` 能力。

### share
- 选项：`policy`（start、end、race、retry，默认 start）、`retries`、`interval`（retry 策略用）、`methods`（默认 GET、HEAD）。请求级（optIn）：`true`、策略枚举、`DiomanShareOptions`、`false`。`ctx.key()` 为 null 时直接放行。
- 模型：
  - start：首发 `next(patch: cancelToken = 共享 token，并去掉 Dio 层超时)`；每个订阅者用 `ctx.wait(共享结果, timeout: 自己的超时)`，各自离开只结束自己；最后一个订阅者离开时才取消共享请求。首发取消不连累跟随者。
  - end：每次都发，只有序号最新的结果落定。
  - race：每次都发，第一个 `r.ok` 的结果胜出，全部失败时用最后一个。
  - retry：共享同一个结果，失败时在内部按 `retries`、`interval` 重试，调用方只看到最终一次。
- 跟随者拿到同一个只读的 `R`。共享以"一次尝试"为单位；retry 在外层，每次重试都会重新进入 share。共享表按策略分开存放。
- 删除：结算登记簿、`hasMultipleDownstreamSettlers`、`registerDownstreamSettler`、`isJoined`。

### cache
- 选项沿用 0.7：`persist`（存储接口）、`cachePolicy`（memo、persist、both）、`expires`、`clone`、`maxEntries`、`shouldCache(request)`（默认 GET、HEAD）、`shouldStore(R r)`（默认 `r.ok`）、`clock`。请求级（optIn）：`true`、`Duration` = expires、`DiomanCacheOptions`、`false`。
- 模型：方法不允许、key 为 null、或带 mock 探测标记时直接放行；命中（惰性检查过期）直接返回，不触网；未命中时登记新鲜度屏障，`r = await next()`，屏障允许且 `shouldStore(r)` 才写入，`finally` 回收屏障记录。管理操作 `api.plugin<DiomanCacheApi>()` 提供 `remove`、`clear`；`clear()` 会作废在途的写入。
- 变化：`shouldStore` 默认就是 `r.ok`（归一化之后业务失败天然不会进缓存），不再需要和 normalize 接线。

### offline
- 选项沿用 0.7：`isOnline()`、`onConnectivityChanged`、`shouldQueue(request)`、`maxQueueSize`、`maxWait`。请求级：`false`、`DiomanOfflineOptions`。
- 模型：在线 → `next()`。离线：不排队或队列已满 → `ctx.fail(kind: offline, reason: …)`；排队 → `ctx.wait(恢复在线, timeout: maxWait)`，恢复后 `next()`，超时返回 `kind: offline, reason: 'timeout'`，取消返回 cancel。位于 source 最内层，所以缓存命中在离线时照常返回。
- 删除：`DiomanOfflineException`（改为 `kind: offline`）。

## 6. envs

只有 `setup`，不进链：按 `rules` 选第一个命中的配置写进 `client.dio.options`（写入前克隆，不与规则模板共享容器）。

## 7. 内核服务

- **key**：算法沿用 0.7（深模式递归排序 Map、List 保序、`ignores`、`namespace`），由客户端级 `DiomanKeyOptions` 配置。请求级 `call.key`：省略为自动；字符串为显式 key；`false` 表示不参与 cache/share。在 prepare 阶段之后惰性求值。
- **取消**：`api.cancelAll(reason)` 返回本次中止的逻辑请求数；`api.pending`。
- **debug**：`Dioman(debug: true)` 使用默认日志器；记录调用点、插件告警（例如请求体无法克隆、mock 缺少 `mockUrl`）。

## 8. TokenManager

保留 `DiomanTokenManager` 接口（`accessToken`、`refreshToken`、`setTokens`、`clear` 等），删除 `canRefresh`：是否已判过期的状态收进 auth 的实例内部。

B4 终审补充：

| 缺口 | 裁决 |
|---|---|
| share start 订阅者的"自己的超时"怎么由 Dio 三个阶段超时合成 | 上限 = connect + receive，请求带请求体时再加 send；参与计算的任一项为 null 或 ≤0 就不设上限（与 Dio 对 null/零的语义一致）。不取最小值：connect 5s、receive 30s 时，最小值会把正常的慢下载判成超时 |
| cache 的时钟 | 插件级 `clock` 保留，只管 TTL，省略时用客户端时钟；请求级 `now` 仍删除 |
| cache `remove(key)` 的失效范围 | 只作废该 key 的在途读写，`clear()` 才作废全部；旧版提升全局代次，属行为变化 |
| 持久化条目格式 | 写入时增加 result 元信息（`shouldStore` 允许存失败结果时要保留分类），旧的两字段格式仍可读 |
| offline 判定顺序 | 选项关闭 → 在线直接放行 → 再调用 `shouldQueue`，在线时不调用判定回调 |
| 迁移承接用例 | 标题写 `[API id]` 加行为，不写迁移行号；行号对应只留在报告证据里 |

/// 不可变归一化结果，同时是可预期失败的异常载体。
class DiomanResponse<D> implements Exception {
  /// 创建已分类的结果；插件通常通过 ctx.fail 构造。
  const DiomanResponse(
      {required this.ok,
      required this.kind,
      required this.status,
      required this.httpStatus,
      this.code,
      this.message = _emptyMessage,
      required this.data,
      this.reason,
      required this.request,
      this.raw,
      this.error,
      this.errorStack,
      this.callSite});

  /// 是否成功。
  final bool ok;

  /// 结果类别。
  final DiomanKind kind;

  /// 插件读取的逻辑状态。
  final int status;

  /// 原始 HTTP 状态，未收到为零。
  final int httpStatus;

  /// 原始业务码。
  final Object? code;

  /// 结果消息。
  final String message;

  /// 已拆包的数据。
  final D? data;

  /// 插件失败的细分原因。
  final String? reason;

  /// 产生此结果的请求。
  final DiomanRequest request;

  /// Dio 原始响应。
  final Response<dynamic>? raw;

  /// 首次程序异常或传输异常。
  final Object? error;

  /// 原始异常栈。
  final StackTrace? errorStack;

  /// debug 模式下的调用点。
  final StackTrace? callSite;

  /// 派生指定数据类型及调用点；原对象不会被修改。
  DiomanResponse<T> copy<T>({StackTrace? callSite}) => DiomanResponse<T>(
      ok: ok,
      kind: kind,
      status: status,
      httpStatus: httpStatus,
      code: code,
      message: message,
      data: ok ? data as T? : (data is T ? data as T : null),
      reason: reason,
      request: request,
      raw: raw,
      error: error,
      errorStack: errorStack,
      callSite: callSite ?? this.callSite);

  /// 安全描述只包含标量，避免 raw、data 或 error 的循环引用及自定义 toString。
  @override
  String toString() => 'DiomanResponse(${kind.name}, $status, $message)';
}

/// 响应判定服务；钩子只在收到适用响应时执行。
class DiomanResponseFactory<R extends DiomanResponse<dynamic>> {
  /// 定制信封与判定，例如 `DiomanResponseFactory(envelope: null)`。
  const DiomanResponseFactory(
      {this.envelope = const Envelope(),
      this.isEnvelope,
      this.isSuccess,
      this.logicalStatus,
      this.expectJson = false,
      this.create});

  /// null 关闭信封识别。
  final Envelope? envelope;

  /// 自定义信封识别，call.envelope=false 优先。
  final bool Function(Map<dynamic, dynamic>)? isEnvelope;

  /// 只对 2xx 信封调用的成功判定。
  final bool Function(Object?)? isSuccess;

  /// 收到响应后的逻辑状态映射。
  final int Function(DiomanResponse<dynamic>)? logicalStatus;

  /// 声明 JSON 但仍是非空字符串时判 parse。
  final bool expectJson;

  /// 构造扩展结果；抛错时内核使用基础结果，避免再次调用坏钩子。
  final R Function(DiomanResponse<dynamic>)? create;

  /// 经扩展构造器创建结果；坏工厂回退 internal 并保留首次程序异常。
  DiomanResponse<dynamic> _finish(DiomanResponse<dynamic> init) {
    try {
      return create == null ? init : create!(init);
    } catch (e, s) {
      return _internal(init.request, init.error ?? e, init.errorStack ?? s);
    }
  }

  /// 带调用点派生结果并重新经过扩展构造器；不修改共享结果。
  DiomanResponse<dynamic> withCallSite(
          DiomanResponse<dynamic> result, StackTrace callSite) =>
      _finish(result.copy<dynamic>(callSite: callSite));

  /// 用 [data] 派生 [result] 并经扩展工厂；例如缓存克隆保持业务元信息。
  DiomanResponse<dynamic> derive(
          DiomanResponse<dynamic> result, Object? data) =>
      _finish(DiomanResponse<dynamic>(
          ok: result.ok,
          kind: result.kind,
          status: result.status,
          httpStatus: result.httpStatus,
          code: result.code,
          message: result.message,
          data: data,
          reason: result.reason,
          request: result.request,
          raw: result.raw,
          error: result.error,
          errorStack: result.errorStack,
          callSite: result.callSite));

  /// 直接构造不依赖用户回调的 internal。
  static DiomanResponse<dynamic> _internal(
          DiomanRequest request, Object error, StackTrace stack) =>
      DiomanResponse(
          ok: false,
          kind: DiomanKind.internal,
          status: 0,
          httpStatus: 0,
          data: null,
          request: request,
          error: error,
          errorStack: stack);

  /// 将 [failure] 经过扩展点创建结果；例如 `factory.fail(request, failure)`。
  DiomanResponse<dynamic> fail(DiomanRequest request, DiomanFailure failure) =>
      _finish(DiomanResponse(
          ok: failure.kind == DiomanKind.ok,
          kind: failure.kind,
          status: failure.status,
          httpStatus: 0,
          message: failure.message,
          data: failure.data,
          request: request,
          reason: failure.reason,
          error: failure.error,
          errorStack: failure.errorStack));

  /// 分类响应或异常；[cancelled] 的优先级最高，任何钩子异常都归 internal。
  DiomanResponse<dynamic> normalize(DiomanRequest request,
      {Response<dynamic>? raw,
      Object? error,
      StackTrace? stack,
      bool cancelled = false}) {
    try {
      if (cancelled) {
        return fail(
            request,
            DiomanFailure(
                kind: DiomanKind.cancel, error: error, errorStack: stack));
      }
      if (error != null) {
        var kind = DiomanKind.internal;
        var status = 0;
        if (error is DioException) {
          switch (error.type) {
            case DioExceptionType.connectionTimeout:
            case DioExceptionType.sendTimeout:
            case DioExceptionType.receiveTimeout:
              kind = DiomanKind.timeout;
            case DioExceptionType.connectionError:
            case DioExceptionType.badCertificate:
              kind = DiomanKind.network;
            case DioExceptionType.badResponse:
              kind = DiomanKind.http;
              status = error.response?.statusCode ?? 0;
            case DioExceptionType.unknown:
              final cause = error.error;
              if (cause is SocketException ||
                  cause is HttpException ||
                  cause is TlsException) {
                kind = DiomanKind.network;
              }
            case DioExceptionType.cancel:
              return fail(
                  request,
                  DiomanFailure(
                      kind: DiomanKind.cancel,
                      error: error,
                      errorStack: stack));
            default:
              break;
          }
        }
        final cause = error is DioException ? error.error ?? error : error;
        return _finish(DiomanResponse(
            ok: false,
            kind: kind,
            status: status,
            httpStatus: status,
            data: null,
            request: request,
            error: cause,
            errorStack: error is DioException
                ? error.stackTrace
                : stack ?? StackTrace.current,
            raw: error is DioException ? error.response : null));
      }
      final response = raw!;
      final http = response.statusCode ?? 0;
      final httpOk = http >= 200 && http < 300;
      final body = response.data;
      if (httpOk &&
          expectJson &&
          body is String &&
          body.isNotEmpty &&
          (response.headers.value(Headers.contentTypeHeader) ?? _emptyMessage)
              .toLowerCase()
              .contains('json')) {
        return _finish(DiomanResponse(
            ok: false,
            kind: DiomanKind.parse,
            status: http,
            httpStatus: http,
            data: body,
            request: request,
            raw: response));
      }
      final env = envelope;
      final wrapped = request.call.envelope != false &&
          env != null &&
          body is Map &&
          (isEnvelope?.call(body) ??
              (body.containsKey(env.code) &&
                  (body.containsKey(env.data) ||
                      body.containsKey(env.message))));
      final code = wrapped ? (body)[env.code] : null;
      final ok = httpOk &&
          (!wrapped ||
              (isSuccess?.call(code) ??
                  (code == null || code == 0 || code == '0')));
      final status = !ok && httpOk && code is int && code >= 400 && code <= 599
          ? code
          : http;
      var result = DiomanResponse<dynamic>(
          ok: ok,
          kind:
              ok ? DiomanKind.ok : (httpOk ? DiomanKind.biz : DiomanKind.http),
          status: status,
          httpStatus: http,
          code: code,
          message: wrapped
              ? '${(body)[env.message] ?? _emptyMessage}'
              : _emptyMessage,
          data: wrapped ? (body)[env.data] : body,
          request: request,
          raw: response);
      if (logicalStatus != null) {
        result = DiomanResponse(
            ok: result.ok,
            kind: result.kind,
            status: logicalStatus!(result),
            httpStatus: http,
            code: code,
            message: result.message,
            data: result.data,
            request: request,
            raw: response);
      }
      return _finish(result);
    } catch (e, s) {
      return _internal(request, e, s);
    }
  }
}

/// 全部 HTTP 状态交给工厂，不让 Dio 构造 badResponse。
bool _acceptStatus(int? _) => true;

/// 递归克隆 JSON 请求体；FormData 使用 Dio 的 clone，其他类型保留身份。
Object? _clone(Object? value, DiomanLogger log) {
  if (value is Map) return value.map((k, v) => MapEntry(k, _clone(v, log)));
  if (value is List) return value.map((v) => _clone(v, log)).toList();
  if (value is FormData) return value.clone();
  if (value != null && value is! num && value is! bool && value is! String) {
    log('请求体无法克隆，回退原引用');
  }
  return value;
}

/// 执行一次发送；传输构造、Dio 及工厂异常都在本边界归一化。
Future<DiomanResponse<dynamic>>
    diomanTransport<R extends DiomanResponse<dynamic>>(
        Dio dio, DiomanContext<R> ctx) async {
  try {
    final cancelled = ctx.cancelled();
    if (cancelled != null) return cancelled;
    final options = ctx.request.toRequestOptions()
      ..validateStatus = _acceptStatus
      ..cancelToken = ctx.cancelToken;
    if (ctx.replayable) options.data = _clone(options.data, ctx.log);
    final raw = await dio.fetch<dynamic>(options);
    return ctx.factory
        .normalize(ctx.request, raw: raw, cancelled: ctx.isCancelled);
  } catch (e, s) {
    return ctx.factory
        .normalize(ctx.request, error: e, stack: s, cancelled: ctx.isCancelled);
  }
}

const _absent = Object();

/// 有效 Dio 配置的不可变请求视图；入口合并一次后构造。
class DiomanRequest {
  /// 从 [options] 拷贝有效配置；[call] 保存调用选项，例如 `DiomanRequest(ro)`。
  DiomanRequest(RequestOptions options, {DiomanCall call = const DiomanCall()})
      : _options = options.copyWith(),
        data = options.data,
        headers = Map.unmodifiable(options.headers),
        query = Map.unmodifiable(options.queryParameters),
        extra = Map.unmodifiable(options.extra),
        call = DiomanCall(
            key: call.key,
            envelope: call.envelope,
            silent: call.silent,
            options: Map.unmodifiable(call.options));

  /// 完整有效配置的私有副本，永不交给 Dio 修改。
  final RequestOptions _options;

  /// 请求体引用；不透明载荷由传输按需克隆，插件通过补丁替换。
  final Object? data;

  /// 不可修改的请求头。
  final Map<String, dynamic> headers;

  /// 不可修改的查询参数。
  final Map<String, dynamic> query;

  /// 不可修改的用户扩展字段。
  final Map<String, dynamic> extra;

  /// 调用配置。
  final DiomanCall call;

  /// 有效路径。
  String get path => _options.path;

  /// 有效方法。
  String get method => _options.method;

  /// 有效基础地址。
  String get baseUrl => _options.baseUrl;

  /// 有效 URI。
  Uri get uri => toRequestOptions().uri;

  /// 连接超时。
  Duration? get connectTimeout => _options.connectTimeout;

  /// 发送超时。
  Duration? get sendTimeout => _options.sendTimeout;

  /// 接收超时。
  Duration? get receiveTimeout => _options.receiveTimeout;

  /// 当前传输令牌；未指定时内核使用逻辑请求令牌。
  CancelToken? get cancelToken => _options.cancelToken;

  /// 通知静默标记。
  bool get silent => call.silent;

  /// 返回可独立修改的有效 Dio 配置，例如 `dio.fetch(r.toRequestOptions())`。
  RequestOptions toRequestOptions() => _options.copyWith(
      headers: Map.from(headers),
      queryParameters: Map.from(query),
      extra: Map.from(extra))
    ..data = data;

  /// 合并 [patch] 并返回独立视图；不修改基准请求。
  DiomanRequest patch(RequestPatch patch) {
    final o = toRequestOptions();
    final h = LinkedHashMap<String, dynamic>.from(o.headers);
    for (final key in patch.removeHeaders) {
      h.removeWhere((k, _) => k.toLowerCase() == key.toLowerCase());
    }
    for (final e in patch.headers.entries) {
      h.removeWhere((k, _) => k.toLowerCase() == e.key.toLowerCase());
      h[e.key] = e.value;
    }
    o.headers = h;
    o.queryParameters.removeWhere((k, _) => patch.removeQuery.contains(k));
    o.queryParameters.addAll(patch.query);
    o.extra.addAll(patch.extra);
    if (!identical(patch.data, _absent)) o.data = patch.data;
    if (patch.path != null) o.path = patch.path!;
    if (patch.method != null) o.method = patch.method!;
    if (!identical(patch.connectTimeout, _absent)) {
      o.connectTimeout = patch.connectTimeout as Duration?;
    }
    if (!identical(patch.sendTimeout, _absent)) {
      o.sendTimeout = patch.sendTimeout as Duration?;
    }
    if (!identical(patch.receiveTimeout, _absent)) {
      o.receiveTimeout = patch.receiveTimeout as Duration?;
    }
    if (!identical(patch.cancelToken, _absent)) {
      o.cancelToken = patch.cancelToken as CancelToken?;
    }
    return DiomanRequest(o, call: call);
  }
}

/// 按方法路径路由、序列和第 n 次失败的假 adapter。
class FakeNetwork implements HttpClientAdapter {
  /// 创建空路由 adapter，缺省返回成功信封。
  FakeNetwork({this.onSend});

  /// 发送已记录时通知测试，避免猜测微任务次数。
  final void Function(FakeSend send)? onSend;

  /// 路由表，例如 GET:/pets。
  final routes = <String, List<FakeReply>>{};

  /// 每个路由的已发送次数。
  final counts = <String, int>{};

  /// 第 n 次发送覆盖脚本。
  final failOn = <int, Object>{};

  /// 所有实际发送。
  final sends = <FakeSend>[];

  /// 是否已关闭。
  bool closed = false;

  /// 执行假网络发送，读取 Dio 已转换的线上请求体。
  @override
  Future<ResponseBody> fetch(RequestOptions options,
      Stream<Uint8List>? requestStream, Future<void>? cancelFuture) async {
    final bytes = requestStream == null
        ? <int>[]
        : await requestStream.expand((b) => b).toList();
    Object? body;
    if (bytes.isNotEmpty) {
      final text = utf8.decode(bytes, allowMalformed: true);
      try {
        body = jsonDecode(text);
      } catch (_) {
        body = text;
      }
    }
    sends.add(FakeSend(options, body, options.headers[invariantRequestHeader]));
    final sendIndex = sends.length;
    onSend?.call(sends.last);
    final route = '${options.method}:${options.path}';
    final index = counts.update(route, (n) => n + 1, ifAbsent: () => 1) - 1;
    final replies = routes[route] ?? const [FakeReply()];
    final reply = replies[index < replies.length ? index : replies.length - 1];
    if (reply.gate != null) await reply.gate;
    final error = failOn[sendIndex] ?? reply.error;
    if (error != null) throw error;
    return ResponseBody.fromString(jsonEncode(reply.body), reply.status,
        headers: {
          Headers.contentTypeHeader: [Headers.jsonContentType]
        });
  }

  /// 记录 adapter 被关闭。
  @override
  void close({bool force = false}) => closed = true;
}

- [P2] Keep detached competing transfers cancellable during disposal — E:\workspaces\dart-labs-pipeline\dioman\lib\src\pipeline\plugins\share.dart:232-239
  With `end` or `race`, a winning response can finish every subscriber while another transport remains pending. Those logical requests are then removed from the client's pending registry, and the shared entry is detached, so neither `cancelAll()` nor `dispose()` can cancel the remaining transport. Reproducing this with a blocked older `end` request leaves its transport token uncancelled after disposal. Track retained transports independently until completion so lifecycle cancellation still reaches them.

- [P2] Clone explicitly cached failures before exposing their data — E:\workspaces\dart-labs-pipeline\dioman\lib\src\pipeline\plugins\cache.dart:421-421
  When `shouldStore` accepts a failed response, the cache stores and returns the original response without applying the configured clone policy. Mutating the first caller's failure data therefore changes subsequent cache hits, even with `clone: DiomanClonePolicy.deep`. For example, changing a cached business failure's `{n: 1}` payload to `{n: 9}` makes the next request receive `{n: 9}`. Apply cloning to stored failures as well as successes.
